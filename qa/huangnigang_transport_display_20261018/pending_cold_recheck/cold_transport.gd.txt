extends SceneTree
## Fresh-process consumer of the real slot produced by the rendered fixture.
var checks: Array = []
var failures: Array = []
var trusted: Dictionary = {}
var restored: Variant = null
var writer_pid := 0
var slot_before := ""
var slot_after := ""
var expected_raw_ids: Array = []
var actual_ids: Array = []
var normalized_expected: Array = []
var normalized_actual: Array = []

func _initialize() -> void:
	run.call_deferred()

func check(label: String, passed: bool) -> bool:
	checks.append({"label": label, "passed": passed})
	if not passed:
		failures.append(label)
		print("FAIL ", label)
	return passed

func run() -> void:
	writer_pid = int(OS.get_environment("HG_PRODUCER_PID"))
	if not check("new engine process differs from completed writer", writer_pid > 0 and OS.get_process_id() != writer_pid):
		finish(); return
	var profile: String = OS.get_environment("HG_PRIVATE_PROFILE").replace("\\", "/").simplify_path()
	var safe: bool = profile.is_absolute_path() and DirAccess.dir_exists_absolute(profile)
	for key in ["APPDATA", "LOCALAPPDATA", "TEMP", "TMP"]:
		var current: String = OS.get_environment(key).replace("\\", "/").simplify_path()
		safe = safe and current.to_lower() == (profile + "/" + key.to_lower()).to_lower()
	if not check("same four private environment roots, Steam disabled", safe and OS.get_environment("STEAM_DISABLED") == "1" and OS.get_environment("CAMPAIGN_QA") == "1"):
		finish(); return
	var file := FileAccess.open(OS.get_environment("HG_COLD_EXPECTED"), FileAccess.READ)
	var expected: Dictionary = JSON.parse_string(file.get_as_text()); file.close()
	if not check("writer completed all three real Session installations", expected.passed and expected.session_roundtrips.size() == 3):
		finish(); return
	trusted = load("res://scripts/run_content_identity.gd").new().resolve_runtime_identity()
	if not check("new process resolves same real executed input identity", trusted.get("ok", false) and trusted.get("source_mode", false) and trusted.get("save_eligible", false) and trusted.content_version == expected.trusted_identity.content_version):
		print("IDENTITY ", trusted); finish(); return
	var pack: Dictionary = load("res://scripts/run_level1_world_factory.gd").prepare_runtime(trusted)
	if not check("new process prepares installed runtime", pack.get("ok", false)):
		print("RUNTIME ", pack); finish(); return
	var store: Variant = load("res://scripts/run_slot_store.gd").new("user://hg_transport_world/v1")
	var read: Dictionary = store.read_slot()
	if not check("original writer slot independently readable", read.get("ok", false)):
		print("SLOT ", read); finish(); return
	slot_before = read.file_sha256
	var last: Dictionary = expected.session_roundtrips[-1]
	if not check("exact writer bytes retained across process exit", slot_before == last.saved_sha256):
		finish(); return
	var menu: Variant = load("res://scenes/menu.tscn").instantiate()
	root.add_child(menu); current_scene = menu
	await process_frame
	paused = true
	var session: Variant = load("res://scripts/run_world_session.gd").new(trusted, pack.runtime, "user://hg_transport_world/v1")
	var prepared: Dictionary = session.prepare_restore(menu)
	if not check("fresh process prepares saved complete world", prepared.get("ok", false)):
		print("PREPARE ", prepared); session.dispose(); finish(); return
	var installed: Dictionary = await session.commit_restore_async()
	if not check("fresh process actually installs complete world", installed.get("ok", false)):
		print("INSTALL ", installed); session.dispose(); finish(); return
	restored = installed.battle
	check("actual installed short Huangnigang script", restored.level.get_script() == load("res://scripts/levels/level1_huangnigang_short.gd"))
	expected_raw_ids = last.eleven_entity_ids
	actual_ids = restored.level.bundles.map(func(u): return u.entity_id)
	var expected_integral: bool = expected_raw_ids.size() == 11 and expected_raw_ids.all(func(value): return (typeof(value) == TYPE_INT or typeof(value) == TYPE_FLOAT) and is_finite(float(value)) and float(value) == floor(float(value)) and value > 0 and value < 2147483647)
	check("writer JSON IDs are exactly integral and bounded", expected_integral)
	# Godot JSON numbers are floats; retain original values and compare explicit
	# decimal identities only after validating that normalization loses no data.
	if expected_integral:
		normalized_expected = expected_raw_ids.map(func(value): return str(int(value)))
		normalized_actual = actual_ids.map(func(value): return str(value))
	check("all eleven original entity IDs preserved", expected_integral and actual_ids.size() == 11 and normalized_actual == normalized_expected)
	print("COLD_ID_COMPARISON raw_expected=", expected_raw_ids, " actual=", actual_ids, " canonical_expected=", normalized_expected, " canonical_actual=", normalized_actual)
	check("all seven carts delivered, weighted count eleven", restored.level.delivered == 11 and restored.level._delivered_cart_indices(restored).size() == 7 and restored.level.cargo.is_empty())
	check("restored regroup stage remains paused", restored.level.st == restored.level.WITHDRAW and restored.phase == restored.Phase.FIGHT and paused)
	check("all seven loaded flags including cart index six restored", restored.level.cargo_ready.size() == 7 and restored.level.cargo_ready.get(6, false))
	check("two-load and one-load death attempts retained", restored.level.force_attempts.get(1, 0) == 1 and restored.level.force_attempts.get(6, 0) == 1)
	var visible_primary := 0
	var stowed_secondary := 0
	for i in range(11):
		var unit: Variant = restored.level.bundles[i]
		if i < 7:
			if restored.units.has(unit) and unit.visible: visible_primary += 1
		else:
			if not restored.units.has(unit) and not unit.visible and unit.process_mode == Node.PROCESS_MODE_DISABLED and unit.get_parent() == restored.units_root:
				stowed_secondary += 1
	check("seven ground carts and four packed secondary loads", visible_primary == 7 and stowed_secondary == 4)
	check("original empty date carts do not reappear", restored.level.jujube_carts.size() == 7 and restored.level.jujube_carts.all(func(cart): return not cart.visible))
	for i in range(7):
		check("compact cart %d display source survives restart" % i, restored.level.bundles[i].display_name == "第%d车·%d担" % [i + 1, 2 if i < 4 else 1])
	var reread: Dictionary = store.read_slot()
	slot_after = reread.file_sha256 if reread.get("ok", false) else ""
	check("original slot bytes unchanged after cold installation", slot_after == slot_before)
	session.dispose()
	finish()

func finish() -> void:
	var report := {"passed": failures.is_empty(), "checks": checks, "failures": failures,
		"expected_raw_entity_ids": expected_raw_ids, "actual_entity_ids": actual_ids, "normalized_expected_ids": normalized_expected, "normalized_actual_ids": normalized_actual,
		"writer_pid": writer_pid, "consumer_pid": OS.get_process_id(), "trusted_identity": trusted,
		"slot_sha256_before": slot_before, "slot_sha256_after": slot_after,
		"actual_restored_battle": restored != null, "checkpoint": "seven carts eleven loads delivered",
		"active_carrier_checkpoint_cold_tested": false, "ordinary_player_continue_entry_tested": false,
		"full_story_tested": false, "native_steam_tested": false, "new_package_tested": false}
	var file := FileAccess.open(OS.get_environment("HG_COLD_OUT"), FileAccess.WRITE)
	file.store_string(JSON.stringify(report, "\t")); file.close()
	print("HG_COLD_REPORT ", checks.size(), " checks; failures ", failures.size())
	quit(0 if failures.is_empty() else 1)
