# Preserve every untouched line's original mixed line ending for reviewable Git
# diffs. Candidate and isolated native input still remain byte-identical.
old_lines=(root/'scripts/hud.gd').read_bytes().decode('utf-8').splitlines(keepends=True)
new_lines=h.splitlines()
preserved=[]
for op,i,j,a,b in difflib.SequenceMatcher(None,[x.rstrip('\r\n') for x in old_lines],new_lines,autojunk=False).get_opcodes():
    if op=='equal': preserved.extend(old_lines[i:j])
    elif op in ('insert','replace'):
        neighbor=old_lines[min(i,len(old_lines)-1)]
        ending='\r\n' if neighbor.endswith('\r\n') else '\n'
        preserved.extend(line+ending for line in new_lines[a:b])
assert ''.join(preserved).replace('\r\n','\n')==h
(out/'hud.gd').write_bytes(''.join(preserved).encode('utf-8'))
