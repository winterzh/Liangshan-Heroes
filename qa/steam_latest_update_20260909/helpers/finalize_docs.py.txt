def require(condition: bool, message: str) -> None:
    if not condition:
        raise RuntimeError(message)


def sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def json_bytes(value: object) -> bytes:
    return (json.dumps(value, ensure_ascii=False, indent=2) + "\n").encode("utf-8")


def file_sha(path: Path) -> str:
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def git_read(*args: str) -> str:
    return subprocess.check_output(
        ["git", "-C", str(ROOT), *args], encoding="utf-8"
    ).strip()


def member_fields(item: dict) -> tuple[str, int, str]:
    require(isinstance(item, dict), "server member must be an object")
    name = item.get("path", item.get("name"))
    size = item.get("bytes", item.get("size"))
    digest = item.get("sha1")
    require(isinstance(name, str) and Path(name).name == name,
            "server member must have a plain file name")
    require(type(size) is int and size >= 0, "server member size missing/invalid")
    require(isinstance(digest, str) and re.fullmatch(r"[0-9a-fA-F]{40}", digest),
            "server member SHA1 missing/invalid")
    return name, size, digest.lower()

    rule = b"qa/steam_latest_update_20260909/** -text -whitespace"
    require(rule not in attrs.splitlines(), "QA attributes already added")
    old_files[attr_path] = attrs
    new_files[attr_path] = attrs.rstrip(b"\r\n") + b"\n" + rule + b"\n"

    input_hashes = {
        "source_head": SOURCE_HEAD,
        "runtime_commit": RUNTIME_COMMIT,
        "inputs": [
            {"path": "publication_receipt.json", "bytes": len(receipt_data),
             "sha256": sha256(receipt_data)},
            {"path": "candidate_delivery.json", "bytes": len(delivery_data),
             "sha256": DELIVERY_SHA},
            {"path": "copy.json", "bytes": len(copy_data), "sha256": COPY_SHA},
            {"path": COVER_REL, "bytes": cover_path.stat().st_size, "sha256": COVER_SHA},
        ],
        "copy_field_lengths": lengths,
        "candidate_zip_and_all_six_members_reverified": True,
        "server_six_members_match": True,
        "preexisting_document_hashes": [
            {"path": path.relative_to(ROOT).as_posix(), "bytes": len(data), "sha256": sha256(data)}
            for path, data in sorted(old_files.items())
        ],
        "helper_sha256": file_sha(Path(__file__)),
    }
    new_files[QA / "finalization_input_hashes.json"] = json_bytes(input_hashes)

    # Recheck preimages immediately before applying the full reviewed output set.
    for path, original in old_files.items():
        require(path.read_bytes() == original, f"document changed during preparation: {path}")
    for path in new_files:
        if path not in old_files:
            require(not path.exists(), f"new destination appeared: {path}")
    for path, data in new_files.items():
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_bytes(data)
        require(path.read_bytes() == data, f"document write mismatch: {path}")

    inventory = []
    for path in sorted(QA.rglob("*")):
        require(not path.is_symlink() and not path.is_junction(), "QA contains linked path")
        if path.is_file() and path.name != "evidence_manifest.json":
            inventory.append({"path": path.relative_to(QA).as_posix(),
                              "bytes": path.stat().st_size, "sha256": file_sha(path)})
    manifest_data = {"source_head": SOURCE_HEAD, "build_id": build,
                     "file_count": len(inventory), "files": inventory}
    manifest_path = QA / "evidence_manifest.json"
    manifest_path.write_bytes(json_bytes(manifest_data))
    require(json.loads(manifest_path.read_bytes()) == manifest_data, "manifest write mismatch")
    for item in inventory:
        path = QA / item["path"]
        require(path.stat().st_size == item["bytes"] and file_sha(path) == item["sha256"],
                f"final evidence changed: {item['path']}")
    print(json.dumps({"complete": True, "build_id": build, "manifest_id": manifest,
                      "event_id": event, "qa_file_count": len(inventory),
                      "written_paths": [path.relative_to(ROOT).as_posix() for path in new_files]
                      + [manifest_path.relative_to(ROOT).as_posix()]}, ensure_ascii=False))


if __name__ == "__main__":
    main()
