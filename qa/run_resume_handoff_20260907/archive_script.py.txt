ROOT = Path(__file__).resolve().parent.parent

sha = lambda x: hashlib.sha256(x).hexdigest()
def read(p): return json.loads(p.read_text(encoding='utf-8'))
def need(x, why):
    if not x: raise RuntimeError(why)
need(not QA.exists(), 'Never overwrite handoff')
need(not (ROOT / '.godot/redraw_rejection_source.lock').exists(), 'Own engine lock remains')
planned, source_rows, runs = {}, [], []
secret = re.compile(rb'-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----|gh[pousr]_[A-Za-z0-9]{30,}|AKIA[0-9A-Z]{16}|sk-proj-[A-Za-z0-9_-]{40,}')
def add(name, destination=None, expected=None):
    p = ROOT / name
    need(p.is_file() and not p.is_symlink() and p.resolve().is_relative_to(ROOT.resolve()), 'Unsafe path ' + name)
    need(not any(x in {'private_profile', 'profile', '__pycache__', '.godot'} for x in p.relative_to(ROOT).parts), 'Private/cache path')
    need(not p.name.startswith(('players_', 'identity_players_')), 'Player listing')
    raw = p.read_bytes()
    need(len(raw) < 10 * 1024 * 1024 and secret.search(raw) is None, 'Size/secret review')
    raw.decode('utf-8')
    need(p.suffix.lower() not in {'.exe', '.dll', '.pck', '.zip', '.png', '.jpg', '.pem', '.key'}, 'Binary excluded')
    need(expected is None or sha(raw) == expected, 'Hash differs: ' + name)
    destination = destination or 'drafts/' + name + '.txt'
    target = QA / destination
    need(target.resolve().is_relative_to(QA.resolve()), 'Destination escaped')
    need(destination not in planned or planned[destination]['raw'] == raw, 'Destination conflict')
    planned[destination] = {'raw': raw, 'original': name}
    return target.relative_to(ROOT).as_posix()

# Raw source evidence can reuse committed, byte-identical earlier QA blobs.
reuse = {}
def rows(value):
    if isinstance(value, dict):
        digest, path = value.get('sha256'), value.get('archive') or value.get('archive_path')
        if isinstance(digest, str) and isinstance(path, str): yield digest, path
        for child in value.values(): yield from rows(child)
    elif isinstance(value, list):
        for child in value: yield from rows(child)
index_name = 'qa/run_gameplay_rng_integration_20260907/source_index.json'
for digest, name in rows(read(ROOT / index_name)):
    p = ROOT / name
    if p.is_file() and p.resolve().is_relative_to((ROOT / 'qa').resolve()) and sha(p.read_bytes()) == digest:
        reuse.setdefault(digest, name)
def source(name, digest, role):
    if digest in reuse:
        archive = reuse[digest]
        raw = (ROOT / archive).read_bytes()
    else:
        raw = (ROOT / name).read_bytes()
        archive = add(name, 'sources/' + digest + '/' + Path(name).name + '.txt', digest)
    need(sha(raw) == digest, 'Source differs')
    source_rows.append({'source':name,'archive':archive,'sha256':digest,'bytes':len(raw),'role':role})
