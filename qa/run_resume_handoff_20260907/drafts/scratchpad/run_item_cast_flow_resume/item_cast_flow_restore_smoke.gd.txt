func _initialize() -> void:
	call_deferred("_run")

func _check(name: String, passed: bool) -> bool:
	checks.append({"label": name, "passed": passed})
	if not passed: failed += 1
	return passed

func _ok(name: String, result: Dictionary) -> bool:
	checks.append({"label": name, "passed": bool(result.get("ok", false)), "code": String(result.get("code", "")), "field": String(result.get("field", ""))})
	if not result.get("ok", false):
		failed += 1
		return false
	return true

func _source_guard(label: String) -> void:
	for path in manifest.source_sha256:
		_check(label + " " + String(path), FileAccess.get_sha256(String(path)) == manifest.source_sha256[path])

func _finish(aborted: bool = false) -> void:
	for graph in graphs: graph.dispose()
	graphs.clear()
	for object in owned:
		if is_instance_valid(object): object.free()
	paused = false
	if manifest_ready: _source_guard("source after")
	var failures: Array = []
	for row in checks:
		if not row.passed: failures.append(row.label)
	var report: Dictionary = {"suite": "item-cast-flow-restore", "run_id": manifest.get("run_id", ""),
		"complete": not aborted, "passed": failed == 0 and not aborted, "checks": checks,
		"check_count": checks.size(), "failures": failures, "failed_count": failed,
		"process_id": OS.get_process_id(), "actual_user_dir": OS.get_user_data_dir(),
		"source_sha256": manifest.get("source_sha256", {}), "observed_steps": observed_steps,
		"scope": "Only pending/walking item queues through real creators, consumers, Unit phases and Inventory timers. The fixture directly supplies Unit and inventory fields, trusted QA items, a real empty map, a trusted fixture UID maximum and staged arrivals. No full UnitGraph/Inventory loader, whole-graph UID validator, movement simulation, existing visual restoration, full Battle resume, RNG isolation, PCK or menu continuation."}
	if not report_path.is_empty():
		var file: FileAccess = FileAccess.open(report_path, FileAccess.WRITE)
		if file == null:
			quit(1)
			return
		file.store_string(JSON.stringify(report, "\t"))
		file.close()
	print("[item-cast-flow-restore QA] ", JSON.stringify(report))
	quit(0 if report.passed else 1)

func _record_with_payload(record: Dictionary, payload: Dictionary, codec: Variant) -> Dictionary:
	var copy: Dictionary = record.duplicate(true)
	copy["payload"] = codec.encode(payload).value
	return copy

func _has_caster(rows: Array, unit: Variant, field: String) -> bool:
	for row in rows:
		if row[field] == unit: return true
	return false

func _step(battle: Variant, units: Array, step: int) -> void:

		units[6].order_move(Vector2(80, 900))
		units[7].cancel_cast_windup()

		# This fixture supplies arrival; it does not claim complete movement replay.
		units[0].position = units[10].position - Vector2(40, 0)
		units[1].position = units[11].position - Vector2(40, 0)
	for unit in units:
		if unit.inventory == null: continue
		if unit._cast_t > 0.0:
			unit._physics_process(0.125)
		else:
			unit.inventory.tick(0.125)
	battle._walk_item_cast_pass(0.125)
	battle._tick_pending_item_casts()

func _copy_value(value: Variant) -> Variant:
	return value.duplicate(true) if typeof(value) in [TYPE_ARRAY, TYPE_DICTIONARY] else (value.duplicate() if typeof(value) == TYPE_PACKED_VECTOR2_ARRAY else value)

func _unit_states(units: Array) -> Array:
	var result: Array = []
	for unit in units:
		var state: Dictionary = {}
		for field in UNIT_FIXTURE_FIELDS: state[field] = _copy_value(unit.get(field))
		state["position"] = unit.position
		if unit.inventory != null:
			var inv: Dictionary = {}
			for field in INVENTORY_FIXTURE_FIELDS: inv[field] = _copy_value(unit.inventory.get(field))
			state["inventory"] = inv
		result.append(state)
	return result

func _live_uid_max(units: Array) -> int:
	var maximum: int = 0
	for unit in units:
		if unit.inventory == null: continue
		for item in unit.inventory.slots:
			if not item.is_empty(): maximum = maxi(maximum, item.uid)
	return maximum
