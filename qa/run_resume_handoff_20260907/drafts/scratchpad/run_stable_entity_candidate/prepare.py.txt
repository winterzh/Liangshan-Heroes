def sha(raw): return hashlib.sha256(raw).hexdigest()


def write(path, raw):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(raw)


def replace(text, old, new, count=1):
    assert text.count(old) == count, (old[:120], text.count(old), count)
    return text.replace(old, new)


def method(text, name, transform):
    marker = 'func ' + name + '('
    starts = [m.start() for m in re.finditer(r'^' + re.escape(marker), text, re.M)]
    assert len(starts) == 1, name
    start = starts[0]
    end_match = re.search(r'^func ', text[start + len(marker):], re.M)
    end = start + len(marker) + end_match.start() if end_match else len(text)
    before = text[start:end]
    after = transform(before)
    assert before != after, name
    return text[:start] + after + text[end:]
