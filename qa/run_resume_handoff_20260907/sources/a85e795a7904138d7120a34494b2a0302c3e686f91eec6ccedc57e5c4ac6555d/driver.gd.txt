## Actual engine callback-boundary diagnostic, not a production RunSession.
const Core := preload("res://scratchpad/run_entity_clock_migration/entity_clock_core.gd")
var checks: Array = []
var owned: Array = []
var manifest: Dictionary = {}
var manifest_ready: bool = false
var report_path: String = ""
var runtime_identity: Dictionary = {}
var events: Array = []
var step_core: Variant
var probes: Array = []
var victims: Dictionary = {}
var capture_pending := false
var capture_done := false
var paused_frames := 0
var captured_tick := -1
var closing := false
var callback_counts: Dictionary = {}
var codec: Variant
var battle_script: Variant
var map_script: Script
var unit_script: Script

func _initialize() -> void:
	call_deferred("_run")

func _check(label: String, passed: bool) -> bool:
	checks.append({"label": label, "passed": passed})
	return passed

func _ok(label: String, value: Dictionary) -> bool:
	checks.append({"label": label, "passed": value.get("ok") == true,
		"code": String(value.get("code", "")), "field": String(value.get("field", ""))})
	return value.get("ok") == true

func _source_guard(label: String) -> bool:
	var result: bool = true
	for path in manifest.source_sha256:
		result = _check(label + " " + String(path), FileAccess.get_sha256(String(path)) == manifest.source_sha256[path]) and result
	return result

func _finish(aborted: bool = false) -> void:
	for object in owned:
		if is_instance_valid(object): object.free()
	paused = false
	if manifest_ready: _source_guard("source after")
	var failures: Array = []
	for row in checks:
		if not row.passed: failures.append(row.label)
	var report: Dictionary = {"suite": "frame-boundary-probe", "run_id": manifest.get("run_id", ""),
		"complete": not aborted, "passed": not aborted and failures.is_empty(),
		"checks": checks, "check_count": checks.size(), "failures": failures,
		"process_id": OS.get_process_id(), "actual_user_dir": OS.get_user_data_dir(),
		"source_sha256": manifest.get("source_sha256", {}),
		"engine_binary_sha256": manifest.get("engine_binary_sha256", ""),
		"rendering_method": RenderingServer.get_current_rendering_method(),
		"runtime_identity": runtime_identity, "events": events,
		"scope": "Instrumented SceneTree callback ordering with real physics timers, tweens, deferred calls and queue_free. Actual candidate clock core and provider. Not a Battle integration, real input test or complete save/continue."}
	if not report_path.is_empty():
		var file: FileAccess = FileAccess.open(report_path, FileAccess.WRITE)
		if file == null:
			quit(1)
			return
		file.store_string(JSON.stringify(report, "\t")); file.close()
	print("[frame-boundary-probe QA] ", JSON.stringify(report))
	quit(0 if report.passed else 1)

func _run() -> void:
	var path: String = OS.get_environment("RUN_RESTORE_QA_MANIFEST")
	var data: Variant = JSON.parse_string(FileAccess.get_file_as_string(path)) if not path.is_empty() else null
	if typeof(data) != TYPE_DICTIONARY:
		_check("host manifest", false); _finish(true); return
	manifest = data
	for key in ["run_id", "private_user", "report", "engine_binary_sha256"]:
		if typeof(manifest.get(key)) != TYPE_STRING or manifest[key].is_empty():
			_check("host manifest " + key, false); _finish(true); return
	if typeof(manifest.get("source_sha256")) != TYPE_DICTIONARY or manifest.source_sha256.is_empty():
		_check("host source manifest", false); _finish(true); return
	report_path = manifest.report
	if not report_path.is_absolute_path() or FileAccess.file_exists(report_path):
		report_path = ""; _check("fresh absolute report", false); _finish(true); return
	manifest_ready = true
	if not _check("private user directory", OS.get_user_data_dir().replace("\\", "/").simplify_path().to_lower() == manifest.private_user.replace("\\", "/").simplify_path().to_lower()):
		_finish(true); return
	if not _source_guard("source before"):
		_finish(true); return
	var provider_script: Script = load("res://scripts/run_content_identity.gd")
	var provider: Variant = provider_script.new()
	var identity: Dictionary = provider.resolve_runtime_identity()
	if not _ok("actual provider identity", identity):
		_finish(true); return
	if not _check("actual provider save eligibility and native engine", identity.get("save_eligible") == true and identity.get("engine_binary_sha256") == manifest.engine_binary_sha256):
		_finish(true); return
	runtime_identity = {"content_version": identity.content_version,
		"source_mode": identity.get("source_mode"), "identity_scope": identity.get("identity_scope"),
		"engine_binary_sha256": identity.engine_binary_sha256}
	battle_script = load("res://scripts/battle.gd")
	map_script = load("res://scripts/game_map.gd")
	unit_script = load("res://scripts/unit.gd")
	var codec_script: Script = load("res://scripts/run_state_value_codec.gd")
	var layer_script: Script = battle_script.FogLayer
	var codec_script: Script = load("res://scripts/run_state_value_codec.gd")
	step_core = Core.new(codec_script, identity.content_version)
	# This fresh process deliberately limits rendering to expose catch-up physics.
	Engine.max_fps = 10
	for index in range(2):
		var probe := StepProbe.new()
		probe.host = self; probe.index = index
		probe.process_physics_priority = index
		root.add_child(probe); owned.append(probe); probes.append(probe)
	physics_frame.connect(_physics_boundary)
	process_frame.connect(_process_boundary)
	_record("probe armed", 0)

class StepProbe:
	extends Node
	var host: Variant
	var index: int
	func _physics_process(_delta: float) -> void:
		host.node_callback(index)

func _record(kind: String, tick: int) -> void:
	events.append({"kind": kind, "tick": tick, "physics": Engine.get_physics_frames(),
		"process": Engine.get_process_frames(), "paused": paused,
		"in_physics": Engine.is_in_physics_frame()})

func _late(kind: String, tick: int) -> void:
	_record(kind, tick)
	if kind == "timer": call_deferred("_late", "timer nested deferred", tick)

func node_callback(index: int) -> void:
	var tick: int = step_core.simulation_tick()
	_record("node " + str(index), tick)
	callback_counts[tick] = int(callback_counts.get(tick, 0)) + 1
	_check("node runs in open step " + str(tick) + ":" + str(index), step_core.step_open() and not paused)
	if index != 0: return
	call_deferred("_late", "node deferred", tick)
	var victim := Node.new()
	root.add_child(victim); victims[tick] = victim
	victim.queue_free()
	var timer: SceneTreeTimer = create_timer(0.0, false, true, false)
	timer.timeout.connect(_late.bind("timer", tick))
	var tween: Tween = create_tween()
	tween.set_process_mode(Tween.TWEEN_PROCESS_PHYSICS)
	tween.tween_callback(_late.bind("tween", tick))
	if tick == 8:
		capture_pending = true
		_record("capture requested in node callback", tick)
		_check("mid callback capture refused", step_core.capture([]).get("code") == "CAPTURE_DURING_STEP")

func _commit(kind: String) -> void:
	if not step_core.step_open(): return
	var tick: int = step_core.simulation_tick()
	var kinds: Array = []
	for event in events:
		if event.tick == tick: kinds.append(event.kind)
	_check("both node callbacks finished " + str(tick), callback_counts.get(tick) == 2)
	for required in ["node deferred", "timer", "tween", "timer nested deferred"]:
		_check(required + " before commit " + str(tick), required in kinds)
	_check("queued deletion before commit " + str(tick), victims.has(tick) and not is_instance_valid(victims[tick]))
	_ok("commit core " + str(tick), step_core.commit_after_callback_barrier())
	_record("commit at " + kind, tick)

func _physics_boundary() -> void:
	if closing or step_core == null: return
	_commit("physics boundary")
	if capture_pending:
		_pause_for_capture()
	if paused: return
	_ok("begin core step", step_core.begin_step())
	_record("begin", step_core.simulation_tick())

func _pause_for_capture() -> void:
	if capture_done: return
	paused = true
	captured_tick = step_core.simulation_tick()
	_record("capture barrier paused", captured_tick)
	_ok("committed core capture", step_core.capture([]))
	capture_done = true

func _process_boundary() -> void:
	if closing or step_core == null: return
	_commit("process boundary")
	if capture_pending: _pause_for_capture()
	if not capture_done: return
	paused_frames += 1
	if paused_frames < 3: return
	_check("paused world does not open or advance tick", captured_tick == 8 and step_core.simulation_tick() == 8 and not step_core.step_open())
	_check("paused world has exactly sixteen node callbacks", callback_counts.size() == 8 and callback_counts.values().all(func(value): return value == 2))
	var per_render: Dictionary = {}
	for event in events:
		if event.kind == "begin": per_render[event.process] = int(per_render.get(event.process, 0)) + 1
	var catch_up := false
	for count in per_render.values():
		if count > 1: catch_up = true
	_check("actual multiple physics steps within rendered frame", catch_up)
	closing = true
	physics_frame.disconnect(_physics_boundary); process_frame.disconnect(_process_boundary)
	_finish()
