def save(name,s):

        for op,i,j,a,b in difflib.SequenceMatcher(None,[x.rstrip('\r\n') for x in old],new,autojunk=False).get_opcodes():
            if op=='equal': result.extend(old[i:j])

                e='\r\n' if old[min(i,len(old)-1)].endswith('\r\n') else '\n'
