"""Serial actual release EXE checks with isolated profiles and source/player guards."""
from pathlib import Path
import datetime, hashlib, json, os, re, subprocess, sys, uuid

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / 'tools'))
from contracts.run_content_identity_20260907.probe_runner import utilities
util = utilities()
guard = util.load_helper(util.GUARD, util.GUARD_SHA, 'upload_source_guard')
safety = util.load_helper(util.HELPER, util.HELPER_SHA, 'upload_process_guard')
candidate = Path(sys.argv[1]).resolve()
candidate.relative_to((ROOT / '.godot/steam_candidates').resolve())
proof = json.loads((candidate / 'receipt.json').read_text(encoding='utf-8'))
assert proof['complete']
exe = candidate / 'windows/LiangshanHeroes.exe'
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
assert sha(exe) == next(r['sha256'] for r in proof['outputs'] if r['path'] == exe.name)
run = Path(__file__).parent / ('smoke_' + datetime.datetime.now().strftime('%Y%m%d_%H%M%S'))
run.mkdir(exist_ok=False)
profile = Path('<PRIVATE_PROFILE_ROOT>') / ('upload_smoke_' + uuid.uuid4().hex[:8])
profile.mkdir(exist_ok=False)
real_user = Path(os.environ['APPDATA']) / 'Godot/app_userdata' / guard.project_name(ROOT)
lock = ROOT / '.godot/redraw_rejection_source.lock'
safety.require_exclusive_godot()
source_before = guard.source_receipt(ROOT)
players_before = util.tree(real_user)
receipt = {'complete': False, 'source_head': proof['source_head'], 'exe_sha256': sha(exe), 'cases': []}
with lock.open('x', encoding='utf-8') as file: file.write(str(run))
child = None
try:
    cases = [(f'level{i}', {'SMOKE_TEST':'1','LEVEL':str(i)}, f'[smoke] level{i}') for i in range(1,9)]
    cases += [('defense', {'SMOKE_TEST':'1','SKIRMISH':'1','DEFENSE_HARD_FIX_TEST':'1'}, '[defense_hard_fix]'), ('cleanup', {'SMOKE_TEST':'1','SKIRMISH':'1','FINAL_CLEANUP_TEST':'1'}, '[final_cleanup]'), ('main_menu', {}, '')]
    for name, overrides, marker in cases:
        env = os.environ.copy()
        for key in list(env):
            if key.endswith(('_TEST','_QA','_AUDIT','_QA_MANIFEST')) or key in {'LEVEL','SKIRMISH','SKIRMISH_AI','SCENARIO','CUSTOM_DEFENSE','ARENA','AUTO_MICRO','AUTOMICRO','SCREENSHOT_DIR'}:
                env.pop(key)
        for key in ('APPDATA','LOCALAPPDATA','TEMP','TMP'):
            private = profile / name / key.lower()
            private.mkdir(parents=True)
            env[key] = str(private)
        env.update(STEAM_DISABLED='1', CONTENT_UPDATE_NO_AUTO='1', **overrides)
        command = [str(exe),'--headless','--max-fps','60','--quit-after','180','--log-file',str(run/(name+'.godot.log'))]
        with (run/(name+'.console.log')).open('xb') as log:
            child = subprocess.Popen(command, cwd=exe.parent, env=env, stdout=log, stderr=subprocess.STDOUT, creationflags=subprocess.CREATE_NO_WINDOW)
            pid = child.pid
            child.wait(timeout=90)
        output = (run/(name+'.console.log')).read_text(encoding='utf-8',errors='replace')
        errors = re.findall(r'^.*(?:SCRIPT ERROR|Parse Error|ERROR:|Failed loading resource|Assertion failed).*$', output, re.M)
        line = next((v for v in output.splitlines() if marker and marker in v), '')
        passed = child.returncode == 0 and not errors and (not marker or bool(line))
        if name in ('defense','cleanup'): passed = passed and 'ALL=true' in line
        row = dict(name=name, pid=pid, exit_code=child.returncode, child_exit_confirmed=True, errors=errors, marker=line, passed=passed)
        receipt['cases'].append(row)
        print(json.dumps(row,ensure_ascii=False),flush=True)
        assert passed, row
    receipt['complete'] = True
except BaseException as error:
    receipt['error'] = type(error).__name__ + ': ' + str(error)
finally:
    if child is not None and child.poll() is None:
        child.kill(); child.wait(timeout=30)
    try:
        safety.require_exclusive_godot()
        receipt.update(source_unchanged=guard.source_receipt(ROOT)==source_before, players_unchanged=util.tree(real_user)==players_before, exe_unchanged=sha(exe)==receipt['exe_sha256'], child_exit_confirmed=child is None or child.poll() is not None)
        assert all(receipt[k] for k in ('source_unchanged','players_unchanged','exe_unchanged','child_exit_confirmed'))
        assert lock.read_text(encoding='utf-8') == str(run)
        lock.unlink()
        receipt['lock_released'] = True
    except BaseException as error:
        receipt.update(complete=False, guard_error=str(error))
    util.save(run/'receipt.json',receipt)
    print(json.dumps({'run':str(run),**receipt},ensure_ascii=False))
sys.exit(0 if receipt['complete'] else 1)
