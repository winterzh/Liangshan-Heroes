extends RefCounted
## Fixed test instrumentation. Never dispatches signals, callbacks or SDK work.
static func hit(name: String) -> void:
	if OS.get_environment("DISK_STOP_AT") != name: return
	var path := ProjectSettings.globalize_path("res://checkpoint.json")
	var file := FileAccess.open(path, FileAccess.WRITE)
	if file == null: return
	file.store_string(JSON.stringify({"name": name, "pid": OS.get_process_id()}))
	file.flush()
	file.close()
	# The external runner kills this PID. No SceneTree frame or callback pump.
	while true: OS.delay_msec(20)
