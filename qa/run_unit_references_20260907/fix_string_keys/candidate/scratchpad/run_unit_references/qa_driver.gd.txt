func _initialize() -> void: _run.call_deferred()
func check(passed: bool, label: String, group := "behavior") -> void:
	checks.append({"label":label,"passed":passed,"group":group})
	if not passed: failures.append(label)
func _record_result(stage: String, result: Dictionary) -> void:
	# Only transport status, never stringify records, registries or Objects.
	result_evidence.append({"stage":stage,"ok":result.get("ok",false),"code":result.get("code",""),
		"path":result.get("path",""),"field":result.get("field",""),"restore_ready":result.get("restore_ready",false)})
func _string_keys(value: Variant) -> bool:
	if typeof(value) == TYPE_DICTIONARY:
		for key in value:
			if typeof(key) != TYPE_STRING or not _string_keys(value[key]): return false
	elif typeof(value) == TYPE_ARRAY:
		for item in value:
			if not _string_keys(item): return false
	return true
func _notice(_unit) -> void: callbacks += 1
func _story_notice(_unit,_outcome) -> void: callbacks += 1
func _unit():
	var result = unit_script.new()
	nodes.append(result)
	return result
func _cleanup() -> void:
	registry.clear()
	for node in nodes:
		if is_instance_valid(node): node.free()
	nodes.clear()
func _hop(record: Dictionary) -> Dictionary:
	var parser := JSON.new()
	check(parser.parse(JSON.stringify(record,"",false)) == OK,"real JSON parse", "json")
	return parser.data if parser.data is Dictionary else {}
func _repack(record: Dictionary, mutate: Callable) -> Dictionary:
	var decoded: Dictionary = codec.decode(record.references)
	check(decoded.ok,"negative fixture decodes original finite tree", "fixture")
	if not decoded.ok: return {}
	mutate.call(decoded.value)
	var encoded: Dictionary = codec.encode(decoded.value)
	check(encoded.ok,"negative fixture encodes finite tree", "fixture")
	if not encoded.ok: return {}
	var result := record.duplicate(true)
	result.references = encoded.value
	return _hop(result)
func _reject(record: Dictionary, code: String, label: String) -> void:
	var result: Dictionary = adapter.validate_record(record,registry,SUBJECT_ID,CONTENT)
	check(not result.ok and result.code == code and not result.has("references") and not result.has("record") and not result.restore_ready,label,"rejection")
func _capture_reject(subject, supplied: Variant, code: String, label: String) -> void:
	var result: Dictionary = adapter.capture_references(subject,supplied,CONTENT)
	check(not result.ok and result.code == code and not result.has("record") and not result.restore_ready,label,"rejection")

func _run() -> void:
	var manifest_path := OS.get_environment("UNIT_REFERENCES_QA_MANIFEST")
	var parsed: Variant = JSON.parse_string(FileAccess.get_file_as_string(manifest_path))
	if not parsed is Dictionary: push_error("Unit references QA requires private manifest"); quit(2); return
	manifest = parsed
	for field in ["run_id","run_dir","private_user","report","source_sha256"]:
		if not manifest.has(field): push_error("Unit references QA missing manifest field"); quit(2); return
	for field in ["run_id","run_dir","private_user","report"]:
		if typeof(manifest[field]) != TYPE_STRING or manifest[field].is_empty(): push_error("Unit references QA invalid manifest String"); quit(2); return
	var run_dir := String(manifest.run_dir).replace("\\","/").simplify_path().trim_suffix("/")
	var report_path := String(manifest.report).replace("\\","/").simplify_path()
	if not run_dir.begins_with(ProjectSettings.globalize_path("res://scratchpad/run_unit_references/runs/").replace("\\","/")) or report_path != run_dir+"/report.json":
		push_error("Unit references report must remain in its new scratchpad run"); quit(2); return
	check(ProjectSettings.globalize_path("user://").replace("\\","/").trim_suffix("/") == String(manifest.private_user).replace("\\","/").trim_suffix("/"),"parent-isolated actual profile","environment")
	check(typeof(manifest.source_sha256) == TYPE_DICTIONARY and manifest.source_sha256.size() == SOURCE_FILES.size(),"exact six runtime source entries","source")
	if typeof(manifest.source_sha256) != TYPE_DICTIONARY: quit(2); return
	for name in SOURCE_FILES:
		check(manifest.source_sha256.has(name) and FileAccess.get_sha256("res://"+name) == manifest.source_sha256.get(name,""),"runtime source "+name,"source")
	if failures.is_empty():
		unit_script = load("res://scripts/unit.gd")
		var codec_script: GDScript = load("res://scripts/run_state_value_codec.gd")
		var values_script: GDScript = load("res://scratchpad/run_unit_state_adapter/unit_values.gd")
		var references_script: GDScript = load("res://scratchpad/run_unit_references/unit_references.gd")
		check(unit_script != null and codec_script != null and values_script != null and references_script != null,"real Unit and exact runtime scripts load")
		if failures.is_empty():
			codec = codec_script.new()
			result_evidence.append({"stage":"loaded_source_contract","unit_source_lf":unit_script.source_code.replace("\r\n","\n").sha256_text(),
				"codec_source_lf":codec_script.source_code.replace("\r\n","\n").sha256_text(),"values_source_lf":values_script.source_code.replace("\r\n","\n").sha256_text()})
			adapter = references_script.new(unit_script,codec_script,values_script)
			_cases()
	_cleanup()
	for name in SOURCE_FILES:
		check(FileAccess.get_sha256("res://"+name) == manifest.source_sha256.get(name,""),"source unchanged "+name,"source")
	var groups: Dictionary = {}
	for item in checks: groups[item.group] = int(groups.get(item.group,0))+1
	var report := {"schema":1,"run_id":manifest.run_id,"process_id":OS.get_process_id(),"manifest_sha256":FileAccess.get_sha256(manifest_path),
		"checks":checks,"failures":failures,"passed":not checks.is_empty() and failures.is_empty(),"check_groups":groups,"result_evidence":result_evidence,
		"actual_user_dir":ProjectSettings.globalize_path("user://"),"source_sha256":manifest.source_sha256,"coverage":adapter.coverage() if adapter != null else {},
		"restore_ready":false,"battle_resume_tested":false,"cross_process_id_continuity_tested":false}
	var file := FileAccess.open(report_path,FileAccess.WRITE)
	if file == null: push_error("Unit references report open failed"); quit(2); return
	var written: bool = file.store_string(JSON.stringify(report,"\t")+"\n")
	file.flush()
	var write_error := file.get_error()
	file.close()
	if not written or write_error != OK: push_error("Unit references report write failed"); quit(2); return
	print("[unit-references] checks=",checks.size()," failures=",failures.size())
	quit(0 if report.passed else 1)
