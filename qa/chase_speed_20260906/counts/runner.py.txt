from pathlib import Path
import subprocess,os,re,json,hashlib
exe=Path(r'E:\ChatGPT\水浒\godot.local.txt').read_text(encoding='utf8').strip()
records=[]
def run(label,script,headless,overrides):
 env=os.environ.copy()
 for key in ['PERF_MODE','PERF_OUT','PERF_DEFENSE','PERF_LEVEL','PERF_CELL','REDRAW_SCREENSHOT','RTS_TEST_OUT']:
  env.pop(key,None)
 env.update(overrides)
 log=Path('scratchpad/chase-speed-'+label+'.log')
 with log.open('wb') as stream:
  result=subprocess.run([exe]+(['--headless'] if headless else [])+['--path','.','--script','res://'+script],env=env,stdout=stream,stderr=subprocess.STDOUT,timeout=240)
 errors=re.findall(r'SCRIPT ERROR|^ERROR:|\bFAIL\b',log.read_text(encoding='utf8',errors='replace'),re.M)
 records.append({'label':label,'script':script,'headless':headless,'environment':overrides,'exit_code':result.returncode,'error_matches':len(errors)})
 Path('scratchpad/chase_speed_runs.json').write_bytes((json.dumps(records,indent=2)+'\n').encode())
 print(label,result.returncode,'errors',len(errors),flush=True)
 assert result.returncode==0 and not errors,label
run('ranged','tools/ranged_firing_path_test.gd',True,{'RTS_TEST_OUT':'res://.godot/chase_speed_ranged'})
run('ai','tools/ai_schedule_qa.gd',True,{})
run('crowd','tools/crowd_neighbor_qa.gd',True,{})
p=Path('scripts/unit.gd');saved=p.read_bytes()
try:
 marker=b'\t\t# Speed only affects the early give-up window; avoid terrain reads otherwise.'
 assert saved.count(marker)==1
 data=saved.replace(marker,b'\t\tpreload("res://scratchpad/chase_live_counts.gd").observe(can_give_up,_chase_t,CHASE_GIVEUP_FAST,CHASE_GIVEUP)\n'+marker)
 marker=b'func current_move_speed(at := Vector2.INF) -> float:\r\n'
 assert data.count(marker)==1
 data=data.replace(marker,marker+b'\tpreload("res://scratchpad/chase_live_counts.gd").speed_reads += 1\n')
 p.write_bytes(data)
 run('counts','scratchpad/chase_live_counts.gd',False,{'PERF_DEFENSE':'1','PERF_OUT':'res://.godot/chase_speed_counts.json'})
finally:
 p.write_bytes(saved);assert p.read_bytes()==saved
 Path('scratchpad/chase_speed_restoration.json').write_bytes((json.dumps({'source_restored_exactly':True,'path':p.as_posix(),'raw_sha256':hashlib.sha256(saved).hexdigest()},indent=2)+'\n').encode())
run('stress','tools/unit_redraw_stress.gd',False,{'PERF_DEFENSE':'1','PERF_OUT':'res://.godot/chase_speed_stress.json','REDRAW_SCREENSHOT':'res://.godot/chase_speed_stress.png'})
