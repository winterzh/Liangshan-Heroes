extends "res://tools/polish_performance_probe.gd"
const UnitBodyDiag = preload("res://scratchpad/unit_body_sections_diag/generated/ledger.gd")
var unit_sections_output := ""
var unit_sections_mode := ""

func _initialize() -> void:
	unit_sections_output = OS.get_environment("UNIT_BODY_SECTIONS_OUT")
	unit_sections_mode = OS.get_environment("UNIT_BODY_SECTIONS_MODE")
	if unit_sections_output.is_empty() or unit_sections_mode not in ["timed","clockless"]:
		push_error("Explicit Unit body diagnostic output/mode required"); quit(2); return
	var private_root := OS.get_environment("UNIT_BODY_SECTIONS_USER_ROOT").replace("\\","/").simplify_path().trim_suffix("/")
	var actual := ProjectSettings.globalize_path("user://").replace("\\","/").simplify_path()
	var output_root := OS.get_environment("UNIT_BODY_SECTIONS_OUTPUT_ROOT").replace("\\","/").simplify_path().trim_suffix("/")
	var output_clean := unit_sections_output.replace("\\","/").simplify_path()
	if private_root.is_empty() or not actual.to_lower().begins_with(private_root.to_lower()+"/") or output_root.is_empty() or not output_clean.to_lower().begins_with(output_root.to_lower()+"/"):
		push_error("Private child user/output isolation failed"); quit(2); return
	OS.set_environment("POLISH_CASE","defense200")
	OS.set_environment("POLISH_CAMERA","fixed")
	OS.set_environment("POLISH_SECONDS","10")
	OS.set_environment("POLISH_OUT",unit_sections_output.get_base_dir().path_join("m1_10s.json"))
	super._initialize()

func _on_tick() -> void:
	if is_instance_valid(battle_ref): UnitBodyDiag.close_step(false)
	super._on_tick()
	if is_instance_valid(battle_ref): UnitBodyDiag.open_step(physics_tick)

func _unit_sections_physics_boundary() -> void:
	UnitBodyDiag.physics_boundary()

func _unit_sections_process_boundary() -> void:
	UnitBodyDiag.process_boundary()

func _dispose(b) -> void:
	b.process_mode = Node.PROCESS_MODE_DISABLED
	UnitBodyDiag.close_step(true)
	UnitBodyDiag.active = false
	physics_frame.disconnect(_unit_sections_physics_boundary)
	process_frame.disconnect(_unit_sections_process_boundary)
	check(not b._prof_on,"native profiler remains disabled; only complete Unit body sections are timed")
	check(UnitBodyDiag.step_count == physics_tick,"all original M1 ticks including last step collected")
	check(UnitBodyDiag.errors == 0 and not UnitBodyDiag.overflow,"body entry/exit/partition/frame ledger integrity")
	var data := UnitBodyDiag.report()
	data["mode"] = unit_sections_mode
	data["actual_user_dir"] = ProjectSettings.globalize_path("user://")
	var file := FileAccess.open(unit_sections_output,FileAccess.WRITE)
	check(file != null,"Unit sections sidecar opened")
	if file != null:
		var stored: bool = file.store_buffer(JSON.stringify(data,"\t").to_utf8_buffer())
		file.flush()
		var write_error: int = file.get_error()
		file.close()
		check(stored and write_error == OK,"Unit sections sidecar closed after complete write")
	await super._dispose(b)
