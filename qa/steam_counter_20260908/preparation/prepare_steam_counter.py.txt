def rep(s,a,b):
    assert s.count(a)==1,(a,s.count(a)); return s.replace(a,b)
def save(path,s):
    p=root/path; old=p.read_bytes().decode().splitlines(keepends=True); new=s.splitlines(); result=[]
    for op,i,j,a,b in difflib.SequenceMatcher(None,[x.rstrip('\r\n') for x in old],new,autojunk=False).get_opcodes():
        if op=='equal': result.extend(old[i:j])
        elif op in ('insert','replace'):
            ending='\r\n' if old[min(i,len(old)-1)].endswith('\r\n') else '\n'
            result.extend(x+ending for x in new[a:b])
    assert ''.join(result).replace('\r\n','\n')==s
    (out/p.name).write_bytes(''.join(result).encode())
