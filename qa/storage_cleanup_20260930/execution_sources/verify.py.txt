import concurrent.futures
import gzip
import hashlib
import json
from pathlib import Path
import shutil
import subprocess
import sys
import zipfile

ROOT = Path(__file__).resolve().parents[2]
OUT = ROOT/'qa/storage_cleanup_20260930'
PLAN = json.loads((OUT/'plan.json').read_text(encoding='utf-8'))

def sha(path):
    with path.open('rb') as f:
        return hashlib.file_digest(f,'sha256').hexdigest()

def save(name,data):
    (OUT/name).write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')

if sys.argv[1] == 'before':
    rows=[]
    for retained in PLAN['retained_cache_mismatches']:
        for relative in retained['mismatches']:
            p=ROOT/relative
            rows.append({'path':relative,'bytes':p.stat().st_size,'sha256':sha(p)})
    save('retained_unique_cache.json',rows)
    print('Retained differing cache files:',len(rows),sum(r['bytes'] for r in rows))
    sys.exit(0)

with gzip.open(OUT/'protected_before.json.gz','rt',encoding='utf-8') as f:
    rows=json.load(f)
rows+=json.loads((OUT/'retained_unique_cache.json').read_text(encoding='utf-8'))
def check(row):
    p=ROOT/row['path']
    if not p.is_file(): return {'path':row['path'],'error':'missing'}
    if p.stat().st_size!=row['bytes'] or sha(p)!=row['sha256']:return {'path':row['path'],'error':'changed'}
    return None
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
    failures=[r for r in pool.map(check,rows) if r]
remaining=[r['path'] for t in PLAN['targets'] for r in t['files'] if (ROOT/r['path']).exists()]
archive=Path(PLAN['archive']['path'])
assert sha(archive)==PLAN['archive']['sha256']
verified_archive=0
with zipfile.ZipFile(archive) as z:
    assert z.testzip() is None
    for t in PLAN['targets']:
        if t['kind']=='verified_archive':
            for row in t['files']:
                with z.open(row['path']) as f:
                    assert hashlib.file_digest(f,'sha256').hexdigest()==row['sha256']
                verified_archive+=1
receipt=json.loads((OUT/'deletion_receipt.json').read_text(encoding='utf-8-sig'))
assert receipt['completed']
assert receipt['removed_files']==sum(len(t['files']) for t in PLAN['targets'])
assert receipt['removed_bytes']==sum(r['bytes'] for t in PLAN['targets'] for r in t['files'])
result={'passed':not failures and not remaining,'protected_files':len(rows),'protected_bytes':sum(r['bytes'] for r in rows),'failures':failures,'deleted_paths_still_present':remaining,'archive_payload_files_verified':verified_archive,'logical_removed_bytes':receipt['removed_bytes'],'archive_bytes':archive.stat().st_size,'logical_net_before_audit_metadata':receipt['removed_bytes']-archive.stat().st_size,'disk_free_before':PLAN['free_bytes_before'],'disk_free_after':shutil.disk_usage(ROOT).free,'tracked_status':subprocess.check_output(['git','diff','--name-only'],cwd=ROOT,text=True).splitlines(),'original_head':PLAN['head'],'current_head':subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip()}
save('verification.json',result)
print(json.dumps(result,ensure_ascii=False))
assert result['passed']
