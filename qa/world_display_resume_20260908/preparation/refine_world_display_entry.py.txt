def module(s):
 s=rep(s,'\treturn _codec.encode(raw)','\treturn encode_record(raw)')
 s=rep(s,'\tvar decoded: Dictionary = _codec.decode(record)','\tvar decoded: Dictionary = decode_record(record)')
 s+='''
func encode_record(raw: Dictionary) -> Dictionary:
\tvar copy: Dictionary = raw.duplicate(true)
\tvar buffer := ""
\tfor row: Dictionary in copy.children:
\t\tif row.role == "shadow": buffer = row.buffer; row.buffer = ""
\tvar encoded: Dictionary = _codec.encode(copy)
\tif not encoded.ok: return encoded
\t# Fixed f32 bytes are separately bounded, not arbitrary serialized objects.
\treturn {"ok": true, "value": {"payload": encoded.value, "shadow_buffer": buffer}}

func decode_record(record: Variant) -> Dictionary:
\tif not _fields(record, ["payload", "shadow_buffer"]) or typeof(record.shadow_buffer) != TYPE_STRING or record.shadow_buffer.length() > 4096 * 12 * 8: return _bad("DISPLAY_ENVELOPE")
\tvar decoded: Dictionary = _codec.decode(record.payload)
\tif not decoded.ok: return decoded
\tif not _fields(decoded.value, ["schema", "world", "units", "children"]) or typeof(decoded.value.children) != TYPE_ARRAY: return _bad("DISPLAY_RECORD")
\tvar count := 0
\tfor row: Variant in decoded.value.children:
\t\tif typeof(row) != TYPE_DICTIONARY: return _bad("DISPLAY_CHILD_RECORD")
\t\tif row.get("role") == "shadow":
\t\t\tif row.get("buffer") != "": return _bad("DISPLAY_BUFFER_PLACEHOLDER")
\t\t\trow.buffer = record.shadow_buffer; count += 1
\tif count > 1 or (count == 0 and not record.shadow_buffer.is_empty()): return _bad("DISPLAY_BUFFER_OWNER")
\treturn decoded
'''
 return s
edit('run_world_display_state.gd',module)
