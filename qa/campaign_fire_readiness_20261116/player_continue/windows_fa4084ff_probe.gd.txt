extends "menu_resume.gd"
## Runs the prisoner transport through player admission and settlement.

func _run() -> void:
	stage = OS.get_environment("GAO_MENU_STAGE")
	var profile := OS.get_environment("APPDATA").replace("\\","/")
	if OS.get_environment("STEAM_DISABLED") != "1" or not profile.contains("/gao-player-") \
			or not OS.get_user_data_dir().replace("\\","/").begins_with(profile + "/"):
		quit(2)
		return
	AudioServer.set_bus_mute(0,true)
	var settings = root.get_node("Settings")
	settings.edge_scroll = false
	settings.game_speed = 1.0
	flow = root.get_node("ContinueFlow")
	var deadline := Time.get_ticks_msec() + 60000
	while flow.busy() and flow.phase != flow.Phase.ERROR and Time.get_ticks_msec() < deadline:
		await process_frame
	check(flow.last_result.get("startup_checked",false) and flow.can_launch(),"current startup completes")
	if not failures.is_empty():
		facts["startup"] = flow.last_result
		_finish()
		return
	facts["identity"] = flow._startup_identity.get("content_version","")
	check(flow._startup_identity.get("ok",false) and not flow._startup_identity.get("source_mode",true),"runtime identity is from the release pack")
	facts["compiled_sha"] = flow._startup_identity.get("source_sha256","")
	flow.set_script(load("platform_entry.gd"))
	check(not flow._private_qa_enabled(),"player admission is active in the isolated profile")
	route = "story"
	capture_route = true
	facts["checkpoint_kind"] = "capture"
	root.size = Vector2i(1440,900)
	build_cells = GAO_BUILD_CELLS.duplicate()
	match stage:
		"save": await _save()
		"resume": await _resume()
		"reject": await _reject_old_slot()
		_:
			check(false,"known stage")
			_finish()

func _reject_old_slot() -> void:
	var menu = load("res://scenes/menu.tscn").instantiate()
	root.add_child(menu)
	current_scene = menu
	await process_frame
	flow.operation_finished.connect(_rejected,CONNECT_ONE_SHOT)
	if not _clicked(menu,"ContinueBattle"):
		_finish()

func _rejected(result: Dictionary) -> void:
	facts["terminal_code"] = result.get("code","")
	var previous: Dictionary = JSON.parse_string(FileAccess.get_file_as_string(HANDOFF))
	var slot: Dictionary = load("res://scripts/run_slot_store.gd").new().read_slot()
	var campaign = root.get_node("Campaign")
	check(not result.ok and facts.terminal_code == "LOCAL_RUN_TERMINAL","player menu rejects the settled slot")
	check(slot.get("ok",false) and slot.file_sha256 == previous.sha and campaign.has_story_seal("level5"),
		"settled campaign progress and original slot bytes are preserved")
	facts["progress_sha"] = FileAccess.get_sha256("user://campaign.cfg")
	_clicked(flow._overlay,"CloseError")
	_finish()

func _finish() -> void:
	Engine.time_scale = 1
	if stage == "resume" and failures.is_empty():
		var deadline := Time.get_ticks_msec() + 60000
		while flow.busy() and flow.phase != flow.Phase.ERROR and Time.get_ticks_msec() < deadline:
			await process_frame
		var record: Dictionary = root.get_node("Campaign").level_record("level5")
		facts["campaign_record"] = record
		check(flow.last_result.get("terminal_completed",false) and record.cleared and record.story_complete and record.best_done == 4,
			"natural victory and all four story goals are committed by the settlement coordinator")
		facts["progress_sha"] = FileAccess.get_sha256("user://campaign.cfg")
		await _check_finished_debrief(current_scene)
	print("GAO_MENU_RESULT ",JSON.stringify({"stage":stage,"pid":OS.get_process_id(),
		"checks":checks,"failures":failures,"facts":facts}))
	if failures.is_empty():
		if stage != "save":
			root.get_node("AppLifecycle").request_quit("gao_player_fixture")
	else:
		root.get_node("Sfx").shutdown()
		root.get_node("Music").shutdown()
		quit.call_deferred(1)

func _check_finished_debrief(b) -> void:
	check(b.hud._end_root.visible,"continued victory opens the real settlement panel")
	var localization = root.get_node("Localize")
	var catalog: Dictionary = JSON.parse_string(FileAccess.get_file_as_string("res://assets/localization/catalog.json"))
	var snapshots := {}
	var expected := [["火攻 · %s",["举火并完成接应撤离"]],["座船 · %s",["已失去战力"]],
		["押俘 · %s",["已押抵忠义堂"]]]
	for language in ["zh_CN","zh_TW","en","ja"]:
		localization._formatted_sources.clear()
		localization.set_language(language,false)
		await process_frame
		var lines := PackedStringArray()
		for row in expected:
			var args: Array = row[1] if language == "zh_CN" else row[1].map(func(key): return catalog[key][language])
			lines.append((row[0] if language == "zh_CN" else catalog[row[0]][language]) % args)
		check(lines.size() == 3 and Array(lines).all(func(line): return line in b.hud._end_tally.text),"fire, flagship and prisoner escort in "+language)
		snapshots[language] = lines
	facts["debrief"] = snapshots
