extends "res://tools/level4_world_restore_qa.gd"
func cross_resume() -> void:
	get_tree().paused = true
	var menu: Node = load("res://scenes/menu.tscn").instantiate()
	get_tree().root.add_child(menu); get_tree().current_scene = menu
	await get_tree().process_frame
	session = Session.new(trusted, runtime, "user://level4_after_simulation/v1_recaptured")
	var prepared: Dictionary = session.prepare_restore(menu)
	if not check("visual saved stage prepares", prepared.ok): print(prepared); return
	var installed: Dictionary = await session.commit_restore_async()
	if not check("visual saved stage installs", installed.ok): print(installed); return
	var b: Node = installed.battle
	# Camera framing only, after restore; this is not an exact-camera assertion.
	get_node("/root/Settings").edge_scroll = false
	b.camera.position = b.to_screen(b.level.song.position)
	b.camera.reset_smoothing(); b.camera.force_update_scroll()
	for frame in range(12): await get_tree().process_frame
	await RenderingServer.frame_post_draw
	check("visual reframed screenshot", get_viewport().get_texture().get_image().save_png(OS.get_environment("LSH_LEVEL4_REPORT").get_basename()+".png") == OK)
