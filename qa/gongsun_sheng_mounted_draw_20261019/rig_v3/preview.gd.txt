		var h:=HashingContext.new();h.start(HashingContext.HASH_SHA256);h.update(im.get_data().slice(0,1254*1254*4))

class Stage extends Node2D:

	func vec(a: Array)->Vector2:return Vector2(a[0],a[1])
	func axis_for(d: int)->Vector2:return Vector2(0.866 if d%2==0 else -0.866,0.5 if d<2 else -0.5).normalized()
	func _process(delta: float)->void:
		elapsed+=delta
		var next_mode: String="moving"
		if elapsed<3.2:speed=1.0
		elif elapsed<3.8:speed=1.0-smoothstep(3.2,3.8,elapsed);next_mode="slowing"
		elif elapsed<5.0:speed=0.0;next_mode="hold"
		elif elapsed<5.6:speed=smoothstep(5.0,5.6,elapsed);next_mode="starting"
		else:speed=1.0;next_mode="restart"
		if next_mode!=mode:mode=next_mode;controls.append({"mode":mode,"elapsed":elapsed,"motion_time":motion_time,"speed":speed})
		motion_time+=delta*speed
		var next: int=floori(motion_time/cycle*4.0)%4
		if next!=phase:phase=next;transitions.append({"phase":phase,"elapsed":elapsed,"motion_time":motion_time})
		queue_redraw()
	func hoof_info(d: int,foot: Vector2,angle: float)->Dictionary:
		var row: Dictionary=data.hoof_anchors[d];var left: Vector2=vec(row.left);var right: Vector2=vec(row.right);var sole: Vector2=(left+right)*.5
		var factor: float=data.hoof_scale;var rotation: float=angle-(right-left).angle()
		return {"ankle":foot+(vec(row.attach)-sole).rotated(rotation)*factor,"sole_l":foot+(left-sole).rotated(rotation)*factor,"sole_r":foot+(right-sole).rotated(rotation)*factor,"sole":sole,"rotation":rotation,"scale":factor}
	func legs(d: int,t: float)->Array:
		var result: Array=[];var view: Dictionary=data.views[d];var facing: float=1.0 if d%2==0 else -1.0
		var bob: float=2.0*cos(t/cycle*TAU*2.0)*speed-float(data.neutral_body_rise)*(1.0-speed);var axis: Vector2=axis_for(d)
		for i in range(4):
			var fore: bool=i<2;var near: bool=i%2==0
			var side: String=String(view.near_side) if near else ("L" if view.near_side=="R" else "R")
			var id: String=side+("F" if fore else "H")
			var offset: float=0.0 if id in ["LF","RH"] else 0.5
			var p: float=fposmod(t/cycle+offset,1.0);var supporting: bool=p<stance
			var travel: float=stride*(0.5-p/stance) if supporting else stride*(-0.5+(p-stance)/(1.0-stance))
			var lift: float=0.0 if supporting else sin((p-stance)/(1.0-stance)*PI)*38.0
			travel*=speed;lift*=speed;supporting=supporting or speed==0.0
			var root: Vector2=vec(view.fore_root if fore else view.hind_root)
			if not near:root+=Vector2(facing*24.0,-12.0)
			var rest: Vector2=Vector2(root.x+facing*(12.0 if fore else -15.0),float(view.floor_y)-(0.0 if near else 18.0))
			var foot: Vector2=rest+axis*travel-Vector2(0,lift);root.y+=bob
			var upper: float=120.0 if fore else 130.0;var lower: float=95.0
			var roll: float=0.0 if supporting else facing*.22*sin((p-stance)/(1.0-stance)*PI)*speed
			var hoof: Dictionary=hoof_info(d,foot,roll);var ankle: Vector2=hoof.ankle
			var delta: Vector2=ankle-root;var distance: float=delta.length();var u: Vector2=delta/distance
			var along: float=(upper*upper-lower*lower+distance*distance)/(2.0*distance)
			var height: float=sqrt(maxf(0.0,upper*upper-along*along));var pole: float=-facing if fore else facing
			var joint: Vector2=root+u*along+Vector2(-u.y,u.x)*height*pole
			result.append({"id":id,"fore":fore,"near":near,"root":root,"joint":joint,"foot":foot,"ankle":ankle,"hoof":hoof,"sole_l":hoof.sole_l,"sole_r":hoof.sole_r,"rest":rest,"upper":upper,"lower":lower,"phase":p,"stance":supporting,"lift":lift})
		return result
	func cell(d: int)->Vector2:return Vector2((d%2)*627,floori(float(d)/2.0)*627)
	func paint_source(key: String,d: int,at: Vector2,source_at: Vector2,factor: float,angle: float,tint: Color)->void:
		var polygon:=PackedVector2Array()
		for corner in [Vector2.ZERO,Vector2(627,0),Vector2(627,627),Vector2(0,627)]:polygon.append(at+(corner-source_at).rotated(angle)*factor)
		var pieces: Array[PackedVector2Array]=[polygon]
		for piece in pieces:
			var uv:=PackedVector2Array();var colors:=PackedColorArray()
			for point in piece:uv.append(((point-at).rotated(-angle)/factor+source_at+cell(d))/1254.0);colors.append(tint)
			draw_polygon(piece,colors,uv,textures[key])
	func part(key: String,d: int,a: Vector2,b: Vector2,tint: Color)->void:
		var pair: Array=data.anchors[key][d];var s0: Vector2=vec(pair[0]);var s1: Vector2=vec(pair[1])
		paint_source(key,d,a,s0,(b-a).length()/(s1-s0).length(),(b-a).angle()-(s1-s0).angle(),tint)
	func paint_leg(d: int,leg: Dictionary)->void:
		var tint: Color=Color.WHITE if leg.near else Color(.79,.78,.76,1.0)
		var prefix: String="fore" if leg.fore else "hind"
		paint_source("hoof",d,leg.foot,leg.hoof.sole,leg.hoof.scale,leg.hoof.rotation,tint)
		part(prefix+"_lower",d,leg.joint,leg.ankle,tint);part(prefix+"_upper",d,leg.root,leg.joint,tint)
		max_bone_error=maxf(max_bone_error,maxf(absf(leg.root.distance_to(leg.joint)-leg.upper),absf(leg.joint.distance_to(leg.ankle)-leg.lower)))
	func horse(d: int,t: float,debug: bool)->void:
		var poses: Array=legs(d,t);var axis: Vector2=axis_for(d)
		for leg in poses:
			var shift: float=fposmod(t*stride/(stance*cycle),20.0)
			for j in range(-4,5):draw_circle(leg.rest+axis*(float(j)*20.0-shift),1.8,Color(.42,.45,.38,.65))
			draw_circle(leg.foot+Vector2(0,leg.lift),5.0,Color(.18,.18,.14,.8))
		var source:=Rect2(cell(d),Vector2(627,627));var bob: float=2.0*cos(t/cycle*TAU*2.0)*speed-float(data.neutral_body_rise)*(1.0-speed)
		draw_texture_rect_region(textures.tail,Rect2(Vector2(0,bob),Vector2(627,627)),source)
		for i in [1,3,0,2]:paint_leg(d,poses[i])
		draw_texture_rect_region(textures.body,Rect2(Vector2(0,bob),Vector2(627,627)),source)
		if debug:
			for leg in poses:
				var color:=Color("9be5c5") if leg.stance else Color("e9b37c")
				draw_line(leg.root,leg.joint,color,2.0);draw_line(leg.joint,leg.ankle,color,2.0);draw_line(leg.sole_l,leg.sole_r,Color.WHITE,2.0)
				for point in [leg.root,leg.joint,leg.foot]:draw_circle(point,4.0,color)
				draw_string(ThemeDB.fallback_font,leg.foot+Vector2(7,-5),leg.id,HORIZONTAL_ALIGNMENT_LEFT,-1,17,color)
	func ribbon(a: Vector2,b: Vector2,sag: float)->void:
		var points: Array[Vector2]=[];var high: Array[Vector2]=[];var low: Array[Vector2]=[];var count: int=data.rein.segments
		var control: Vector2=(a+b)*.5+Vector2(0,2.0*sag)
		for i in range(count+1):
			var t: float=float(i)/count;var u: float=1.0-t;points.append(a*u*u+control*2.0*u*t+b*t*t)
		for i in range(count+1):
			var tangent: Vector2=points[mini(i+1,count)]-points[maxi(i-1,0)];var normal:=Vector2(-tangent.y,tangent.x).normalized()*float(data.rein.width_native)*.5
			high.append(points[i]+normal);low.append(points[i]-normal)
		var poly:=PackedVector2Array();var uv:=PackedVector2Array();var colors:=PackedColorArray();var sample: Array=data.rein.sample_region
		for i in range(count+1):poly.append(high[i]);uv.append(Vector2(float(sample[0])+float(sample[2])*i/count,float(sample[1]))/1254.0);colors.append(Color.WHITE)
		for i in range(count,-1,-1):poly.append(low[i]);uv.append(Vector2(float(sample[0])+float(sample[2])*i/count,float(sample[1])+float(sample[3]))/1254.0);colors.append(Color.WHITE)
		draw_polygon(poly,colors,uv,textures[String(data.rein.texture)])
	func draw_state(d: int,t: float)->Dictionary:
		var row: Dictionary=data.arm_views[d];var keys: Array=row.keys
		var clock: float=fposmod(t,float(keys.size())*.8)/.8;var index: int=floori(clock);var u: float=smoothstep(0.0,1.0,clock-float(index))
		var a: Dictionary=keys[index];var b: Dictionary=keys[(index+1)%keys.size()]
		var guard: Vector2=vec(a.guard).lerp(vec(b.guard),u);var angle: float=lerp_angle(float(a.angle),float(b.angle),u)
		var sw: Dictionary=row.sword;var src_guard: Vector2=vec(sw.guard);var src_tip: Vector2=vec(sw.tip);var factor: float=float(data.blade_length)/(src_tip-src_guard).length()
		var rotation: float=angle-(src_tip-src_guard).angle();var grip: Vector2=guard+(vec(sw.grip)-src_guard).rotated(rotation)*factor
		var palm: Vector2=grip
		if index==0:palm=vec(row.rest_hand)*(1.0-u)*(1.0-u)+vec(row.hand_approach_control)*2.0*(1.0-u)*u+grip*u*u
		elif index==9:palm=grip*(1.0-u)*(1.0-u)+vec(row.hand_approach_control)*2.0*(1.0-u)*u+vec(row.rest_hand)*u*u
		elif index==10:palm=vec(row.rest_hand)
		var hand_angle: float=angle+PI*.5;var hand_row: Dictionary=row.hand
		var wrist: Vector2=palm-(vec(hand_row.grip)-vec(hand_row.wrist)).rotated(hand_angle)*float(data.hand_scale)
		var root: Vector2=vec(row.shoulder);var delta: Vector2=wrist-root;var distance: float=delta.length();var direction: Vector2=delta/maxf(distance,.001)
		var upper: float=data.upper_length;var lower: float=data.fore_length;var along: float=(upper*upper-lower*lower+distance*distance)/(2.0*maxf(distance,.001))
		var height: float=sqrt(maxf(0.0,upper*upper-along*along));var facing: float=1.0 if d%2==0 else -1.0
		var elbow: Vector2=root+direction*along+Vector2(-direction.y,direction.x)*height*facing
		var constrained: bool=index in [0,1,2,7,8,9,10]
		return {"index":index,"label":a.label,"guard":guard,"angle":angle,"sword_rotation":rotation,"sword_scale":factor,"grip":grip,"palm":palm,"wrist":wrist,"root":root,"elbow":elbow,"hand_angle":hand_angle,"clip_at_mouth":constrained,"reach_error":maxf(0.0,distance-upper-lower),"tip":guard+(src_tip-src_guard).rotated(rotation)*factor}
	func clip_plane(poly: PackedVector2Array,at: Vector2,normal: Vector2)->PackedVector2Array:
		var result:=PackedVector2Array()
		for i in range(poly.size()):
			var a: Vector2=poly[i];var b: Vector2=poly[(i+1)%poly.size()];var da: float=(a-at).dot(normal);var db: float=(b-at).dot(normal)
			if da>=0.0:result.append(a)
			if (da>=0.0)!=(db>=0.0):result.append(a.lerp(b,da/(da-db)))
		return result
	func sprite_part(key: String,d: int,at: Vector2,source_at: Vector2,factor: float,angle: float,clip_at: Vector2=Vector2.ZERO,clip_normal: Vector2=Vector2.ZERO)->void:
		var poly:=PackedVector2Array()
		for p in [Vector2.ZERO,Vector2(627,0),Vector2(627,627),Vector2(0,627)]:poly.append(at+(p-source_at).rotated(angle)*factor)
		if clip_normal!=Vector2.ZERO:poly=clip_plane(poly,clip_at,clip_normal)
		if poly.size()<3:return
		var uv:=PackedVector2Array();var colors:=PackedColorArray()
		for p in poly:uv.append(((p-at).rotated(-angle)/factor+source_at+cell(d))/1254.0);colors.append(Color.WHITE)
		draw_polygon(poly,colors,uv,textures[key])
	func sleeve(key: String,d: int,at: Vector2,end: Vector2,shift: Vector2)->void:
		var row: Dictionary=data.arm_views[d];var pair: Array=row[key];var a: Vector2=vec(pair[0]);var b: Vector2=vec(pair[1]);var source_axis: Vector2=(b-a).normalized();var source_normal:=Vector2(-source_axis.y,source_axis.x)
		var axis: Vector2=(end-at).normalized();var normal:=Vector2(-axis.y,axis.x);var factor: float=(end-at).length()/(b-a).length();var width: float=.60 if key=="upper_arm" else .67
		var poly:=PackedVector2Array();var uv:=PackedVector2Array();var colors:=PackedColorArray()
		for p in [Vector2.ZERO,Vector2(627,0),Vector2(627,627),Vector2(0,627)]:
			var delta: Vector2=p-a;poly.append(at+shift+axis*delta.dot(source_axis)*factor+normal*delta.dot(source_normal)*factor*width);uv.append((p+cell(d))/1254.0);colors.append(Color.WHITE)
		draw_polygon(poly,colors,uv,textures[key])
	func articulated_rider(d: int,t: float,debug: bool)->void:

		var row: Dictionary=data.arm_views[d];var pose: Dictionary=draw_state(d,t)

		var body_offset: Vector2=vec(row.waist_target)-vec(row.body_waist)*float(data.torso_scale)+shift
		var bit: Vector2=vec(row.bit)+shift;var hand: Vector2=vec(row.rein_hand)*float(data.torso_scale)+body_offset
		ribbon(bit,hand,float(data.rein.sag_native));ribbon(bit+Vector2(3,2),hand+Vector2(2,1),float(data.rein.sag_native)+3.0)

		var sw: Dictionary=row.sword;var clip_at: Vector2=vec(row.mouth)+shift;var out: Vector2=vec(row.out)
		sprite_part("sword",d,pose.guard+shift,vec(sw.guard),pose.sword_scale,pose.sword_rotation,clip_at if pose.clip_at_mouth else Vector2.ZERO,out if pose.clip_at_mouth else Vector2.ZERO)
		var sc: Dictionary=row.scabbard;var source_axis: Vector2=vec(sc.tip)-vec(sc.mouth)
		sprite_part("scabbard",d,vec(row.mouth)+shift,vec(sc.mouth),float(data.scabbard_length)/source_axis.length(),(-out).angle()-source_axis.angle())
		sleeve("upper_arm",d,pose.root,pose.elbow,shift);sleeve("forearm",d,pose.elbow,pose.wrist,shift)

			draw_line(pose.root+shift,pose.elbow+shift,Color("9cdbc9"),1.5);draw_line(pose.elbow+shift,pose.wrist+shift,Color("9cdbc9"),1.5)
			draw_circle(clip_at,3.0,Color("edbc82"));draw_circle(pose.grip+shift,3.0,Color("edbc82"));draw_circle(pose.tip+shift,3.0,Color("edbc82"))
