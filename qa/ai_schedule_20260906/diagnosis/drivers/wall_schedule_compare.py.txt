from pathlib import Path
import os,subprocess,json,hashlib
paths=[Path('scripts/liangshan_entrance.gd'),Path('scripts/liangshan_stockade.gd')]
saved={p:p.read_bytes() for p in paths}
for p,raw in saved.items():assert raw.replace(b'\r\n',b'\n')==subprocess.check_output(['git','show','HEAD:'+p.as_posix()]).replace(b'\r\n',b'\n')
reports=[]
try:
    for label in ['current','legacy','padded_current']:
        for p,raw in saved.items():p.write_bytes(raw)
        if label=='legacy':
            for p in paths:p.write_bytes(subprocess.check_output(['git','show','5eb6e020e98d8cb95aea8af2e9ded647341dd52b:'+p.as_posix()]))
        elif label=='padded_current':
            p=paths[0];raw=saved[p]
            marker=b'\nfunc _add_wall('
            assert raw.count(marker)==1
            # Temporary identity-allocation control, never a production fix.
            raw=raw.replace(marker,b'\n\tfor _padding_index in range(12):\n\t\tadd_child(Node.new())\n'+marker)
            p.write_bytes(raw)
        env=os.environ.copy();env['PERF_DEFENSE']='1';env['PERF_OUT']='res://.godot/wall_schedule_'+label+'.json'
        exe=Path(r'E:\ChatGPT\水浒\godot.local.txt').read_text(encoding='utf8').strip()
        with Path('scratchpad/wall-schedule-'+label+'.log').open('wb') as out:
            result=subprocess.run([exe,'--path','.','--script','res://scratchpad/wall_schedule_probe.gd'],env=env,stdout=out,stderr=subprocess.STDOUT,timeout=120)
        assert result.returncode==0,result.returncode
        report=json.loads(Path('.godot/wall_schedule_'+label+'.json').read_text())
        reports.append({'label':label,**{k:v for k,v in report.items() if k!='raw_frame_ms'}})
        print(label,report['fps'],report['p95_ms'],report['heroes_at_setup'],flush=True)
finally:
    for p,raw in saved.items():p.write_bytes(raw)
    for p,raw in saved.items():assert p.read_bytes()==raw
    Path('scratchpad/wall_schedule_restoration.json').write_bytes((json.dumps({'all_sources_restored':True,'files':[{"path":p.as_posix(),"raw_sha256":hashlib.sha256(raw).hexdigest()} for p,raw in saved.items()]},indent=2)+'\n').encode())
Path('.godot/wall_schedule_comparison.json').write_bytes((json.dumps(reports,indent=2)+'\n').encode())
