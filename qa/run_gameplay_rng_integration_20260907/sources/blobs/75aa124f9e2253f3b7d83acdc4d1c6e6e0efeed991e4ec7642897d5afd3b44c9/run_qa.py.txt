"""Candidate callsite QA. Explicit --freeze after integration, then isolated real processes."""
import argparse
import datetime
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time
import types

sys.dont_write_bytecode = True
HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[1]
BASE = ROOT / 'scratchpad/run_gameplay_rng_r1/run_qa.py'
BASE_SHA = '8f9c82100bf8b635abd74f9a79b31335e72df404f30c62f3f490458af3e0e1ad'
ENTRY = 'res://scratchpad/run_gameplay_rng_callsite_r2/driver.gd'
PHASES = ['matrix', 'startup', 'restore_reject', 'writer', 'reader']

def runtime_names():
    names = ['project.godot']
    names += [p.relative_to(ROOT).as_posix() for directory, suffix in [('scripts', '.gd'), ('scenes', '.tscn')]
              for p in (ROOT / directory).rglob('*' + suffix)]
    names += ['scratchpad/run_gameplay_rng_callsite_r2/driver.gd']
    return sorted(set(names))

def execute(util, safety, exe, out, env):
    safety.require_exclusive_godot()
    command = [str(exe), '--path', str(ROOT), '--headless', '--script', ENTRY]
    process = None
    failure = cleanup = None
    confirmed = False
    started = time.monotonic()
    log_path = out / 'report.log'
    with log_path.open('xb') as log:
        try:
            process = subprocess.Popen(command, cwd=ROOT, env=env, stdout=log,
                                       stderr=subprocess.STDOUT,
                                       creationflags=getattr(subprocess, 'CREATE_NO_WINDOW', 0))
            safety.ACTIVE_GODOT_PROCESS = process
            while process.poll() is None:
                if util.ERROR.search(log_path.read_bytes().decode('utf-8', errors='ignore')):
                    raise RuntimeError('Owned Godot error; see phase report.log')
                if time.monotonic() - started >= 180:
                    raise subprocess.TimeoutExpired(command, 180)
                time.sleep(0.1)
            util.need(process.returncode == 0, 'Owned Godot returned nonzero')
        except BaseException as error:
            failure = error
        finally:
            if process is not None:
                try:
                    if process.poll() is None: process.kill()
                    process.wait(timeout=30)
                    confirmed = process.poll() is not None
                    if confirmed: safety.ACTIVE_GODOT_PROCESS = None
                except BaseException as error:
                    cleanup = type(error).__name__ + ': ' + str(error)
    receipt = {'command': command, 'child_pid': process.pid if process else None,
               'child_started': process is not None, 'child_exit_confirmed': confirmed,
               'exit_code': process.returncode if process else None,
               'wall_seconds': time.monotonic() - started,
               'exception': type(failure).__name__ + ': ' + str(failure) if failure else None,
               'cleanup_error': cleanup}
    util.save(out / 'report_process.json', receipt)
    if failure is not None: raise failure
    util.need(confirmed and cleanup is None, 'Owned child exit unconfirmed')
    safety.require_exclusive_godot()
    return util.read(out / 'report.json'), receipt

def main():
    raw = BASE.read_bytes()
    assert hashlib.sha256(raw).hexdigest() == BASE_SHA, 'Reviewed helpers drifted'
    util = types.ModuleType('callsite_util'); util.__file__ = str(BASE)
    exec(compile(raw.decode(), str(BASE), 'exec'), util.__dict__)
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument('--run', action='store_true')
    mode.add_argument('--freeze', action='store_true', help='Explicitly pin the already integrated production state; no engine')
    parser.add_argument('--godot')
    parser.add_argument('--phase', choices=['matrix', 'startup', 'restore_reject', 'pair', 'all'], default='matrix')
    args = parser.parse_args()
    pins_path = HERE / 'pins.json'
    if args.freeze:
        for path in ['scripts/run_content_identity.gd', 'scripts/run_gameplay_rng.gd', 'scripts/unit.gd']:
            util.need((ROOT / path).is_file(), 'Required integrated production source missing: ' + path)
        util.need('func gameplay_rng_fault(' in (ROOT/'scripts/battle.gd').read_text(encoding='utf-8'), 'Battle candidate not integrated')
        util.need('func lifesteal_frac() -> Dictionary:' in (ROOT/'scripts/unit.gd').read_text(encoding='utf-8'), 'Unit candidate not integrated')
        # Refuse accidentally pinning the pre-R1 Unit. Production may include a
        # root-reviewed further change, but all four required immediate stops remain.
        unit_text = (ROOT/'scripts/unit.gd').read_text(encoding='utf-8').splitlines()
        for call in ['inventory.tick(delta)', 'died.emit(self)', 'pg.leave_garrison()', 'battle.despawn_summon(self)']:
            hits = [i for i, line in enumerate(unit_text) if line.strip().startswith(call)]
            util.need(hits and all(unit_text[i+1].strip() == 'if not _gameplay_rng_fault().is_empty(): return' for i in hits), 'Unit R1 missing: ' + call)
        value = {'frozen': True, 'godot_run': False,
                 'source_sha256': {name: util.file_sha(ROOT/name) for name in runtime_names()},
                 'runner_sha256': util.file_sha(Path(__file__)),
                 'contract_sha256': util.file_sha(HERE/'contract.json')}
        # Explicit refreezes keep the previous source contract as evidence.
        if pins_path.exists():
            history = HERE/'pins_history'
            history.mkdir(exist_ok=True)
            stamp = datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%S%fZ')
            with (history/(stamp+'.json')).open('xb') as old:
                old.write(pins_path.read_bytes())
        pins_path.write_bytes((json.dumps(value, ensure_ascii=False, indent=2)+'\n').encode('utf-8'))
        print(json.dumps({'frozen': True, 'godot_run': False, 'pins_sha256': util.file_sha(pins_path)}))
        return 0
    pins = util.read(pins_path)
    if not pins.get('frozen'):
        util.need(not args.run, 'Pending root candidate integration and explicit --freeze')
        print(json.dumps({'preflight': True, 'frozen': False, 'godot_run': False, 'next': 'Root integrates candidates, then --freeze'}))
        return 0
    hashes = pins['source_sha256']
    util.need(sorted(hashes) == runtime_names(), 'Runtime source path set drifted')
    util.need(all(util.file_sha(ROOT/name) == expected for name, expected in hashes.items()), 'Pinned source drifted')
    util.need(util.file_sha(Path(__file__)) == pins['runner_sha256'], 'Runner drifted')
    util.need(util.file_sha(HERE/'contract.json') == pins['contract_sha256'], 'Contract drifted')
    if not args.run:
        print(json.dumps({'preflight': True, 'frozen': True, 'godot_run': False, 'source_count': len(hashes)}))
        return 0
    util.need(args.godot is not None, 'Actual direct engine required')
    exe = Path(args.godot).resolve()
    util.need(not exe.name.lower().endswith('_console.exe'), 'Pass real non-console engine, not wrapper')
    util.need(util.file_sha(exe) == util.ENGINE_SHA, 'Actual engine mismatch')
    guard = util.load_helper(util.GUARD, util.GUARD_SHA, 'callsite_source_guard')
    safety = util.load_helper(util.HELPER, util.HELPER_SHA, 'callsite_process_safety')
    safety.ROOT = ROOT; safety.ERROR = util.ERROR
    safety.require_exclusive_godot()
    util.need(not util.LOCK.exists(), 'Shared engine lock occupied')
    stamp = datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%S%fZ')
    out = HERE/'runs'/stamp; out.mkdir(parents=True, exist_ok=False)
    token = str(os.getpid()) + '|' + str(out)
    with util.LOCK.open('x', encoding='utf-8') as f: f.write(token)
    phases = PHASES if args.phase == 'all' else ['writer', 'reader'] if args.phase == 'pair' else [args.phase]
    result = {'suite': 'gameplay-rng-callsites', 'complete': False, 'lock_released': False,
              'engine_sha256': util.ENGINE_SHA, 'pins_sha256': util.file_sha(pins_path),
              'production_modified': False, 'source_sha256': hashes, 'phases': [],
              'full_battle_resume_tested': False}
    source = protected = None
    real_user = Path(os.environ['APPDATA'])/'Godot/app_userdata'/guard.project_name(ROOT)
    contract = util.read(HERE/'contract.json')
    try:
        source = guard.source_receipt(ROOT); protected = util.tree(real_user)
        util.save(out/'sources_before.json', source); util.save(out/'players_before.json', protected)
        original_idle = safety.require_exclusive_godot
        def ownership():
            original_idle()
            util.need(util.LOCK.read_text(encoding='utf-8') == token, 'Lock ownership drift')
            util.need(util.tree(real_user) == protected, 'Real player directory drift')
            util.need(util.file_sha(exe) == util.ENGINE_SHA, 'Engine changed')
            util.need(util.file_sha(pins_path) == result['pins_sha256'], 'Pins changed')
            util.need(sorted(hashes) == runtime_names() and all(util.file_sha(ROOT/name) == value for name, value in hashes.items()), 'Runtime changed')
            util.need(util.file_sha(Path(__file__)) == pins['runner_sha256'], 'Runner changed')
            util.need(util.file_sha(HERE/'contract.json') == pins['contract_sha256'], 'Contract changed')
        safety.require_exclusive_godot = ownership
        writer = None
        handoff_sha = None
        writer_report_sha = None
        for phase in phases:
            ownership()
            phase_out = out/phase; phase_out.mkdir()
            profile = phase_out/'private_profile'
            user = profile/'appdata/Godot/app_userdata'/guard.project_name(ROOT)
            user.mkdir(parents=True, exist_ok=False)
            (profile/'localappdata').mkdir(); (profile/'temp').mkdir()
            env, controls = guard.environment(source, phase_out, profile, user)
            for key in list(env):
                if key.startswith(('UNIT_ADAPTER_', 'UNIT_REFERENCES_', 'INVENTORY_QA_', 'BATTLE_FOG_QA_',
                                   'RUN_RESTORE_', 'RUN_RNG_', 'GAMEPLAY_RNG_', 'VALUE_CODEC_', 'STORE_QA_',
                                   'REDRAW_', 'REDUCED_EFFECTS_', 'FIRST_USE_', 'SEPARATION_', 'CHASE_PATH_', 'ANIM_LOAD_')):
                    env.pop(key)
            env.update(TEMP=str(profile/'temp'), TMP=str(profile/'temp'))
            manifest = {'run_id': stamp, 'phase': phase, 'private_user': str(user),
                        'report': str(phase_out/'report.json'), 'source_sha256': hashes,
                        'engine_binary_sha256': util.file_sha(exe), 'handoff': str(out/'handoff.json')}
            if phase == 'reader':
                util.need(writer is not None and util.file_sha(out/'handoff.json') == handoff_sha, 'Writer handoff missing or changed')
                util.need(util.file_sha(out/'writer/report.json') == writer_report_sha, 'Writer report changed')
                manifest.update(handoff_sha256=handoff_sha, writer_pid=writer, writer_report_sha256=writer_report_sha)
            util.save(phase_out/'manifest.json', manifest)
            manifest_sha = util.file_sha(phase_out/'manifest.json')
            env['RUN_RNG_CALLSITE_MANIFEST'] = str(phase_out/'manifest.json')
            util.save(phase_out/'environment.json', {'private_user': str(user), 'controls': controls})
            report, process = execute(util, safety, exe, phase_out, env)
            log = (phase_out/'report.log').read_bytes().decode('utf-8')
            prefix = '[gameplay-rng-callsites QA] '
            marked = [json.loads(line[len(prefix):]) for line in log.splitlines() if line.startswith(prefix)]
            util.need(marked == [report] and not util.ERROR.search(log), 'Strict log/stdout mismatch')
            util.need(report.get('suite') == result['suite'] and report.get('phase') == phase and report.get('run_id') == stamp, 'Wrong suite/run')
            util.need(report.get('process_id') == process['child_pid'], 'Real PID mismatch')
            util.need(Path(report.get('actual_user_dir','')).resolve() == user.resolve(), 'Private user mismatch')
            util.need(report.get('source_sha256') == hashes and report.get('engine_binary_sha256') == util.ENGINE_SHA, 'Source/engine report mismatch')
            checks = report.get('checks', [])
            labels = {row.get('label') for row in checks}
            util.need(report.get('complete') is True and report.get('passed') is True and checks
                      and report.get('failures') == [] and all(row.get('passed') is True for row in checks), 'Failed/incomplete assertions')
            util.need(set(contract['required_labels'][phase]) <= labels, 'Required behavior labels missing')
            util.need(set(contract['always_required']) <= labels, 'Required identity/profile labels missing')
            for name in hashes:
                util.need('source_before:'+name in labels and 'source_after:'+name in labels, 'Incomplete runtime source checks')
            identity = report.get('identity', {})
            util.need(identity.get('ok') is True and identity.get('save_eligible') is True and identity.get('engine_binary_sha256') == util.ENGINE_SHA, 'Untrusted provider result')
            util.need(util.file_sha(phase_out/'manifest.json') == manifest_sha, 'Host manifest changed')
            if phase == 'writer':
                writer = process['child_pid']; handoff_sha = util.file_sha(out/'handoff.json')
                writer_report_sha = util.file_sha(phase_out/'report.json')
                util.need(report['detail'].get('handoff_sha256') == handoff_sha, 'Writer handoff hash mismatch')
            if phase == 'reader':
                util.need(process['child_pid'] != writer and util.file_sha(out/'handoff.json') == handoff_sha, 'Reader process/handoff changed')
                util.need(util.file_sha(out/'writer/report.json') == writer_report_sha, 'Writer report changed during reader')
            result['phases'].append({'phase': phase, 'process_id': process['child_pid'], 'checks': len(checks),
                                     'report_sha256': util.file_sha(phase_out/'report.json')})
        result['complete'] = True
    except BaseException as error:
        result.update(complete=False, error=type(error).__name__ + ': ' + str(error))
    finally:
        try:
            safety.require_exclusive_godot()
            for phase in phases:
                path = out/phase/'report_process.json'
                if path.exists():
                    proc = util.read(path)
                    util.need(not proc['child_started'] or proc['child_exit_confirmed'], 'Owned process exit unconfirmed')
            after = guard.source_receipt(ROOT); players_after = util.tree(real_user)
            util.need(source is not None and after == source, 'Complete production source guard failed')
            util.need(protected is not None and players_after == protected, 'Player guard failed')
            util.need(util.LOCK.read_text(encoding='utf-8') == token, 'Lock ownership changed')
            util.save(out/'sources_after.json', after); util.save(out/'players_after.json', players_after)
            util.LOCK.unlink()
            result.update(lock_released=True, source_unchanged=True, player_unchanged=True)
        except BaseException as error:
            result.update(complete=False, final_guard_error=type(error).__name__ + ': ' + str(error))
        util.save(out/'receipt.json', result)
        print(json.dumps({'run': str(out), 'complete': result['complete'], 'lock_released': result['lock_released']}))
    return 0 if result['complete'] and result['lock_released'] else 1

if __name__ == '__main__': sys.exit(main())
