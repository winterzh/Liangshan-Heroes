def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def once(raw, old, new):
    # Some historical edited blocks have mixed endings. Match exact text under
    # the separately pinned whole-file SHA, changing only the matching span.
    pattern = re.escape(old.encode()).replace(b'\\\n', b'\\r?\\n')
    matches = list(re.finditer(pattern, raw))
    assert len(matches) == 1, repr(old[:140])
    match = matches[0]
    found = match.group()
    replacement = new.encode()
    if found.count(b'\r\n') * 2 >= found.count(b'\n') and b'\n' in found:
        replacement = replacement.replace(b'\n', b'\r\n')
    return raw[:match.start()] + replacement + raw[match.end():]

def function_span(raw, name):
    match = re.search(rb'^func ' + name.encode() + rb'\(', raw, re.M)
    assert match, name
    following = re.search(rb'^(?:func |class )', raw[match.end():], re.M)
    end = match.end() + following.start() if following else len(raw)
    return match.start(), end

def code_only(line):
    # Only used to find the end of these pinned, ordinary call statements.
    # No general GDScript rewriting: inline suites/semicolons are rejected.
    return re.sub(r'"(?:\\.|[^"\\])*"|\'(?:\\.|[^\'\\])*\'|#.*', '', line)

def guard_function(raw, name, calls=(), entry=True, result='', post_result=None):
    start, end = function_span(raw, name)
    lines = raw[start:end].decode().splitlines(keepends=True)
    result_post = result if post_result is None else post_result
    additions = {}
    for i, line in enumerate(lines):
        code = code_only(line)
        if line.startswith('func ') or not any(re.search(r'(?<![\w])' + re.escape(c) + r'\(', code) for c in calls):
            continue
        assert ';' not in code and not re.match(r'\s*(?:if|elif|return|for|while)\b', code), (name, line)
        j, balance = i, 0
        while True:
            part = code_only(lines[j])
            balance += sum(part.count(x) for x in '([{') - sum(part.count(x) for x in ')]}')
            assert balance >= 0, (name, lines[j])
            if balance == 0 and not part.rstrip().endswith('\\'): break
            j += 1
        indent = line[:len(line)-len(line.lstrip('\t'))]
        guard = indent + 'if not _gameplay_rng_issue.is_empty(): return' + (' ' + result_post if result_post else '')
        if j+1 < len(lines) and lines[j+1].rstrip('\r\n') == guard:
            continue
        assert j not in additions, (name, line)
        additions[j] = guard + ('\r\n' if lines[j].endswith('\r\n') else '\n')
        GUARD_RECEIPT.append({'function': name, 'after': ''.join(lines[i:j+1]).strip()})
    if entry:
        header = 0
        while not code_only(lines[header]).rstrip().endswith(':'): header += 1
        guard = '\tif not _gameplay_rng_issue.is_empty(): return' + (' ' + result if result else '')
        if lines[header+1].rstrip('\r\n') != guard:
            assert header not in additions
            additions[header] = guard + ('\r\n' if lines[header].endswith('\r\n') else '\n')
            GUARD_RECEIPT.append({'function': name, 'entry': True})
    changed = ''.join(line + additions.get(i, '') for i, line in enumerate(lines)).encode()
    return raw[:start] + changed + raw[end:]
