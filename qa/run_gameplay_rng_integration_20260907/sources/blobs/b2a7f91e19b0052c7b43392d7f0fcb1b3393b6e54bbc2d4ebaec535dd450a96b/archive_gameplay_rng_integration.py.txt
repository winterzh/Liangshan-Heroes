def need(ok, message):
    if not ok:
        raise RuntimeError(message)

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def read(path):
    return json.loads(Path(path).read_text(encoding="utf-8"))

def relative(path):
    return Path(path).resolve().relative_to(ROOT.resolve()).as_posix()

def under_root(path):
    result = Path(path)
    if not result.is_absolute():
        result = ROOT / result
    result = result.resolve()
    need(result.is_relative_to(ROOT.resolve()), "Outside repository: " + str(path))
    return result

def exact(path, digest=None, size=None):
    path = under_root(path)
    need(path.is_file() and not path.is_symlink(), "Missing regular file: " + str(path))
    raw = path.read_bytes()
    if digest is not None:
        need(sha(raw) == digest, "Source SHA mismatch: " + relative(path))
    if size is not None:
        need(len(raw) == size, "Source size mismatch: " + relative(path))
    return raw

def stdout_report(path, prefix, report):
    rows = [json.loads(line[len(prefix):]) for line in Path(path).read_text(encoding="utf-8").splitlines()
            if line.startswith(prefix)]
    need(len(rows) == 1 and rows[0] == report, "Report/stdout mismatch: " + str(path))

def artifact(source, destination):
    source = under_root(source)
    raw = exact(source)
    need(source.suffix.lower() not in {".exe", ".dll", ".pck", ".zip", ".png", ".jpg", ".webp", ".key", ".pem"},
         "Binary/private artifact denied: " + relative(source))
    need(not any(p in {"private_profile", "profile", "__pycache__"} for p in source.parts)
         and not source.name.startswith(("players_", "identity_players_")), "Player listing denied")
    target = under_root(QA / destination)
    need(target.is_relative_to(QA.resolve()), "Archive target escaped")
    raw.decode("utf-8")
    name = target.relative_to(QA).as_posix()
    if name in planned:
        need(planned[name]["raw"] == raw, "Conflicting archive destination")
    else:
        planned[name] = {"raw": raw, "original": relative(source)}
    return relative(target)

def reusable_rows(value):
    if isinstance(value, dict):
        digest = value.get("sha256")
        path = value.get("archive") or value.get("archive_path") or value.get("archived_path")
        if isinstance(digest, str) and isinstance(path, str):
            yield digest, path
        for child in value.values():
            yield from reusable_rows(child)
    elif isinstance(value, list):
        for child in value:
            yield from reusable_rows(child)

def source(name, role, digest=None, origin=None, private_project=None):
    name = str(name).replace("\\", "/")
    if origin is not None:
        chosen = under_root(origin)
        raw = exact(chosen, digest)
    else:
        options = []
        if digest is not None and (name, digest) in candidates:
            options.append(candidates[(name, digest)])
        options.append(ROOT / name)
        if private_project is not None:
            options.append(private_project / name)
        chosen, raw = None, None
        for option in options:
            if option.is_file():
                trial = exact(option)
                if digest is None or sha(trial) == digest:
                    chosen, raw = option, trial
                    break
        if raw is None and digest in reused:
            chosen = ROOT / reused[digest]
            raw = exact(chosen, digest)
        need(raw is not None, "No exact historical source: " + name)
    digest = sha(raw)
    if digest not in blobs:
        if digest in reused:
            exact(reused[digest], digest, len(raw))
            blobs[digest] = reused[digest]
        else:
            blobs[digest] = artifact(chosen, "sources/blobs/" + digest + "/" + Path(name).name + ".txt")
    record = {"source": name, "archive": blobs[digest], "sha256": digest, "bytes": len(raw),
              "role": role, "original": relative(chosen), "reused_existing": digest in reused}
    mappings.append(record)
    return record

def native_text_sources(rows, label, private_project):

    return result

    need(report["complete"] is True and report["passed"] is expected_passed and len(report["checks"]) == 10

    need(proc["child_started"] is True and proc["child_exit_confirmed"] is True
         and proc["exit_code"] == (0 if expected_passed else 1)
         and report["process_id"] == proc["child_pid"], "Identity process mismatch")

    if expected_passed:
        need(all(c["passed"] is True for c in report["checks"]) and proc["complete"] is True,
             "Incomplete successful identity probe")
        exact(folder / "report.json", proc["report_sha256"])
        need(report["identity"]["provider_sha256"] == provider, "Identity provider mismatch")

        need(report["identity"]["code"] == expected_bad, "Identity failure code changed")

    for path, digest in proc["sources"].items():

            "report_sha256": sha(exact(folder / "report.json")), "actual_entry": entry,

def save_new(path, value):
    raw = (json.dumps(value, ensure_ascii=False, indent=2) + "\n").encode("utf-8")
    with path.open("xb") as handle:
        handle.write(raw)
    return sha(raw)
