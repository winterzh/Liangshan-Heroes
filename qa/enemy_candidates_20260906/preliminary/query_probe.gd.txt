func _timed_query(u,reference: bool,origin: Vector2,faction: int) -> int:
	var elapsed:=0
	for i in range(4000):
		u.position=origin;u.faction=faction;u._target=null;u._chase_best_distance=INF
		var start:=Time.get_ticks_usec()
		if reference:u._old_acquire()
		else:u._new_acquire()
		elapsed+=Time.get_ticks_usec()-start
	return elapsed
