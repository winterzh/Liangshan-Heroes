from pathlib import Path
import os, subprocess, re, json, time, hashlib
exe=Path(r'E:\ChatGPT\水浒\godot.local.txt').read_text(encoding='utf8').strip()
specs=[('enemy','enemy_candidates_qa.gd',True,{},'.godot/enemy_candidates_qa/report.json'),('ai','ai_schedule_qa.gd',True,{},'.godot/ai_schedule_qa/report.json'),('chase','chase_speed_qa.gd',True,{},'.godot/chase_speed_qa/report.json'),('grid','grid_build_qa.gd',True,{},'.godot/grid_build_qa/report.json'),('ranged','ranged_firing_path_test.gd',True,{'RTS_TEST_OUT':'res://.godot/enemy_candidates_ranged'},'.godot/enemy_candidates_ranged/ranged.json'),('stress','unit_redraw_stress.gd',False,{'PERF_DEFENSE':'1','PERF_OUT':'res://.godot/enemy_candidates_stress.json','REDRAW_SCREENSHOT':'res://.godot/enemy_candidates_stress.png'},'.godot/enemy_candidates_stress.json')]
rows=[]
for name,script,headless,overrides,report in specs:
 env=os.environ.copy()
 for key in ['PERF_MODE','PERF_LEVEL','PERF_CELL','PERF_DEFENSE','PERF_OUT','REDRAW_SCREENSHOT','RTS_TEST_OUT']:env.pop(key,None)
 env.update(overrides)
 command=[exe]+(['--headless'] if headless else [])+['--path','.','--script','res://tools/'+script]
 log=Path('scratchpad/enemy-validation-'+name+'.log');start=time.time()
 with log.open('wb') as stream:r=subprocess.run(command,env=env,stdout=stream,stderr=subprocess.STDOUT,timeout=240)
 errors=re.findall(r'^.*(?:SCRIPT ERROR|^ERROR:|\bFAIL\b).*$',log.read_text(encoding='utf8'),re.M)
 data=json.loads(Path(report).read_text(encoding='utf8'))
 row={'name':name,'script':script,'headless':headless,'environment':overrides,'report':report,'log':str(log),'exit_code':r.returncode,'seconds':time.time()-start,'errors':errors,'passed':data.get('passed'),'checks':data.get('checks',len(data.get('mode_checks',[]))),'source_lf_sha256':{p:hashlib.sha256(Path(p).read_bytes().replace(b'\r\n',b'\n')).hexdigest() for p in ['scripts/battle.gd','scripts/unit.gd','tools/'+script]}}
 rows.append(row);Path('scratchpad/enemy_validation_runs.json').write_text(json.dumps(rows,indent=2),encoding='utf8')
 print(json.dumps(row,ensure_ascii=False),flush=True)
 assert r.returncode==0 and not errors
 if name!='stress':assert data['passed'] and not data['failures']
