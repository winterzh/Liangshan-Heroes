def dump(path,data): path.write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n',encoding='utf-8',newline='\n')
def git(*args): return subprocess.check_output(['git',*args],cwd=ROOT)

assert not git('status','--porcelain')
assert sha((OLD/'freeze.json').read_bytes())=='93942811415614d4541a854cc9dee8522c756de05e49f7352470ddad25a37724'

sys.path.insert(0,str(ROOT/'tools')); import run_stabilization_performance as guard
before={r['path']:r for r in guard.tree('b074ba91c28a9e24a32b1ad2469522094f8267cd')}

changed=[n for n in sorted(set(before)|set(after)) if before.get(n)!=after.get(n)]
