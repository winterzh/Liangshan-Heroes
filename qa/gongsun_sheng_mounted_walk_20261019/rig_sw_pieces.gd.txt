	var use_rig: bool=false
	var rig_info: Dictionary={}
	var body_polygons: Array=[]
	func area(poly: PackedVector2Array)->float:
		var result:=0.0
		for i in range(poly.size()):result+=poly[i].cross(poly[(i+1)%poly.size()])*.5
		return absf(result)
	func bone_point(p: Vector2,lower: bool)->Vector2:
		var cfg: Dictionary=data.rig
		if lower:return Vector2(rig_info.knee_new[0],rig_info.knee_new[1])+(p-vec(cfg.knee)).rotated(float(rig_info.lower_rotation))
		return vec(cfg.shoulder)+(p-vec(cfg.shoulder)).rotated(float(rig_info.upper_rotation))
	func prepare_rig()->Dictionary:
		var cfg: Dictionary=data.rig;var shoulder:=vec(cfg.shoulder);var knee:=vec(cfg.knee);var hoof:=vec(cfg.hoof);var target:=vec(cfg.target)
		var l1:=shoulder.distance_to(knee);var l2:=knee.distance_to(hoof);var distance:=shoulder.distance_to(target)
		var cosine:=clampf((l1*l1+distance*distance-l2*l2)/(2*l1*distance),-1,1)
		var knee_new: Vector2=shoulder+Vector2.RIGHT.rotated((target-shoulder).angle()+acos(cosine))*l1
		rig_info={"knee_new":[knee_new.x,knee_new.y],"upper_rotation":(knee_new-shoulder).angle()-(knee-shoulder).angle(),"lower_rotation":(target-knee_new).angle()-(hoof-knee).angle(),"upper_length_error":absf(shoulder.distance_to(knee_new)-l1),"lower_length_error":absf(knee_new.distance_to(target)-l2),"manual_anchor_uncertainty_source_pixels":5}
		var rr: Array=data.textures[String(cfg.texture)].selected_region
		var quad:=PackedVector2Array([Vector2(rr[0],rr[1]),Vector2(rr[0]+rr[2],rr[1]),Vector2(rr[0]+rr[2],rr[1]+rr[3]),Vector2(rr[0],rr[1]+rr[3])])
		var mask:=PackedVector2Array()
		for p in cfg.body_cut:mask.append(vec(p))
		body_polygons=Geometry2D.clip_polygons(quad,mask)
		var kept_area:=0.0;var removed_area:=0.0;var vertices:=0
		for poly in body_polygons:kept_area+=area(poly);vertices+=poly.size()
		for poly in Geometry2D.intersect_polygons(quad,mask):removed_area+=area(poly)
		var edge_error:=0.0
		for key in ["upper_piece","lower_piece"]:
			var pts: Array=cfg[key];vertices+=pts.size()
			for i in range(pts.size()):
				var a:=vec(pts[i]);var b:=vec(pts[(i+1)%pts.size()]);edge_error=maxf(edge_error,absf(a.distance_to(b)-bone_point(a,key=="lower_piece").distance_to(bone_point(b,key=="lower_piece"))))
		var mapped:=bone_point(hoof,true)
		rig_info.merge({"hoof_target_error":mapped.distance_to(target),"mapped_hoof":[mapped.x,mapped.y],"body_area_partition_error":absf(area(quad)-kept_area-removed_area),"body_polygons":body_polygons.size(),"piece_edge_length_error":edge_error,"render_vertices":vertices+4,"source_png_changed":false,"scope":"Rigid painted pieces, clipped original body and unchanged source chest cover; no texture-pixel write. Joint seam/occlusion/quality still requires visual review."})
		return rig_info
	func textured_poly(points: PackedVector2Array,uv_points: PackedVector2Array)->void:
		var mapped:=PackedVector2Array();var uv:=PackedVector2Array();var colors:=PackedColorArray()
		for i in range(points.size()):mapped.append(points[i]*float(data.horse.scale)+vec(data.horse.offset)-vec(data.origin));uv.append(uv_points[i]/1254.0);colors.append(Color.WHITE)
		draw_polygon(mapped,colors,uv,textures[String(data.rig.texture)])
	func paint_rig()->void:
		for poly in body_polygons:textured_poly(poly,poly)
		for key in ["lower_piece","upper_piece"]:
			var points:=PackedVector2Array();var uv:=PackedVector2Array()
			for p in data.rig[key]:uv.append(vec(p));points.append(bone_point(vec(p),key=="lower_piece"))
			textured_poly(points,uv)
		var cover: Array=data.rig.chest_cover;var source:=Rect2(cover[0],cover[1],cover[2],cover[3]);var sc: float=data.horse.scale
		draw_texture_rect_region(textures[String(data.rig.texture)],Rect2(source.position*sc+vec(data.horse.offset)-vec(data.origin),source.size*sc),source)
