extends SceneTree
## Current creators -> real Unit graph + Fx graph + effect arrays -> JSON -> consumers.
## Battle stays off tree: no menu, level deployment, simulation scheduler or disk resume.
const CONTENT := "stable_remaining_visual_lifecycle_v1"
var checks: Array = []
var failures: Array = []
var owned: Array = []
var contexts: Array = []
var manifest: Dictionary = {}
var report_path := ""
var manifest_ready := false
var scripts: Dictionary = {}
var observations: Dictionary = {}

func _initialize() -> void:
	call_deferred("_run")

func _check(label: String, pass_value: bool) -> bool:
	checks.append({"label": label, "passed": pass_value})
	if not pass_value: failures.append(label)
	return pass_value

func _ok(label: String, value: Dictionary) -> bool:
	checks.append({"label": label, "passed": value.get("ok") == true, "code": value.get("code", ""), "field": value.get("field", "")})
	if value.get("ok") != true:
		failures.append(label)
		return false
	return true

func _source_guard(label: String) -> void:
	for path: String in manifest.source_sha256:
		_check(label + " " + path, FileAccess.get_sha256(path) == manifest.source_sha256[path])

func _finish(aborted := false) -> void:
	for context: Variant in contexts: context.dispose()
	for node: Variant in owned:
		if is_instance_valid(node): node.free()
	if manifest_ready: _source_guard("source after")
	var result := {"suite": "mixed-fx-native", "run_id": manifest.get("run_id", ""),
		"passed": failures.is_empty() and not aborted, "complete": not aborted, "failures": failures,
		"checks": checks, "check_count": checks.size(), "failed_count": failures.size(), "observations": observations,
		"process_id": OS.get_process_id(), "actual_user_dir": OS.get_user_data_dir(), "source_sha256": manifest.get("source_sha256", {}),
		"scope": "Eight additional actual timed classes, exact ready and remaining state; paired fixed consumer steps for meteor, ground fire and heal/banner ward. Source and target use explicit same visual test seeds for future creations. Attack-ward next snapshot includes BlinkShotFx; standalone blink lifecycle additionally uses real idle callbacks. No Battle ready, full scheduler, rendering proof, global visual RNG restore, disk slot or production integration."}
	if not report_path.is_empty():
		var file := FileAccess.open(report_path, FileAccess.WRITE)
		if file == null:
			quit(1)
			return
		file.store_string(JSON.stringify(result, "\t"))
		file.close()
	print("[mixed Fx QA] ", JSON.stringify(result))
	quit(0 if result.passed else 1)

func _run() -> void:
	var path := OS.get_environment("RUN_RESTORE_QA_MANIFEST")
	var data: Variant = JSON.parse_string(FileAccess.get_file_as_string(path)) if not path.is_empty() else null
	if typeof(data) != TYPE_DICTIONARY:
		_check("host manifest present", false)
		_finish(true)
		return
	manifest = data
	for key: String in ["run_id", "private_user", "report", "engine_binary_sha256"]:
		if typeof(manifest.get(key)) != TYPE_STRING or manifest[key].is_empty():
			_check("host manifest " + key, false)
			_finish(true)
			return
	if typeof(manifest.get("source_sha256")) != TYPE_DICTIONARY or manifest.source_sha256.is_empty():
		_check("host source manifest", false)
		_finish(true)
		return
	report_path = manifest.report
	if not report_path.is_absolute_path() or FileAccess.file_exists(report_path):
		_check("fresh absolute report path", false)
		report_path = ""
		_finish(true)
		return
	manifest_ready = true
	_check("private profile", OS.get_user_data_dir().replace("\\", "/").simplify_path().to_lower() == manifest.private_user.replace("\\", "/").simplify_path().to_lower())
	_source_guard("source before")
	if not failures.is_empty():
		_finish(true)
		return
	# No preload or game-class annotation in the CLI entry before autoload startup.
	for key: String in ["battle", "unit", "game_map", "hero_inventory", "run_state_value_codec", "run_unit_state", "run_unit_graph", "run_graph_identity", "run_remaining_effect_state", "run_visual_graph", "run_meteor_wards_state", "run_continuous_effect_state", "hud", "projectile", "run_projectile_state", "run_li_brawn_axes_state"]:
		scripts[key] = load("res://scripts/" + key + ".gd")
	paused = true
	root.get_node("Settings").show_damage = true
	for kind: String in ["meteor", "ground_fire", "ground_fire_lite", "ward_heal", "ward_loyalty", "ward_righteous", "arrow_shot_flight", "arrow_shot_impact", "arrow_rain", "flameburst", "ability", "building_collapse", "ward_attack_boundary"]:
		if not await _case(kind):
			_finish(true)
			return
	if not await _blink_engine_case():
		_finish(true)
		return
	if not await _mixed_engine_case():
		_finish(true)
		return
	_finish()

func _shell() -> Variant:
	var b: Variant = scripts.battle.new()
	owned.append(b)
	b.set_block_signals(true)
	b.process_mode = Node.PROCESS_MODE_DISABLED
	b.world = Node2D.new()
	b.add_child(b.world)
	b.map = scripts.game_map.new()
	b.map.init_map(64, 64, "marsh", scripts.game_map.T.GRASS)
	b.map.bake()
	b.world.add_child(b.map)
	b.units_root = Node2D.new()
	b.world.add_child(b.units_root)
	return b

func _stage() -> Node2D:
	var stage := Node2D.new()
	stage.process_mode = Node.PROCESS_MODE_DISABLED
	stage.visible = false
	owned.append(stage)
	root.add_child(stage)
	return stage

func _unit(b: Variant, id: int, faction: int, at: Vector2) -> Variant:
	var u: Variant = scripts.unit.new()
	u.entity_id = id
	u.key = "fixture_" + str(id)
	u.display_name = u.key
	u.visible = false
	u.process_mode = Node.PROCESS_MODE_DISABLED
	u.battle = b
	u.map = b.map
	u.position = at
	u.faction = faction
	u.hp = 1000.0
	u.max_hp = 1000.0
	u.radius = 12.0
	u.base_speed = 120.0
	u._base_speed = 120.0
	u.stance = 3
	b.units_root.add_child(u)
	b.units.append(u)
	b.next_entity_id = id + 1
	return u

func _identity_call(identity: Variant, operation: String, value: Variant) -> Dictionary:
	return identity.call(operation, "_chase_last_id", value)

func _effects(identity: Variant, visuals: Variant) -> Variant:
	return scripts.run_remaining_effect_state.new(scripts.run_state_value_codec, scripts.battle, scripts.unit,
		func(v): return _identity_call(identity, "encode_identity", v),
		func(v): return _identity_call(identity, "validate_identity", v),
		func(v): return _identity_call(identity, "decode_identity", v),
		visuals.encode_token, visuals.validate_token, visuals.decode_token)

func _meteor_state(identity: Variant) -> Variant:
	return scripts.run_meteor_wards_state.new(scripts.run_state_value_codec, scripts.battle, scripts.unit,
		func(v): return _identity_call(identity, "encode_identity", v),
		func(v): return _identity_call(identity, "validate_identity", v),
		func(v): return _identity_call(identity, "decode_identity", v))

func _step(b: Variant, delta: float) -> void:
	b._zone_pass(delta)
	b._ground_dot_pass(delta)
	b._ward_pass(delta)
	for fx: Variant in b.fx_root.get_children():
		if not is_instance_valid(fx) or fx.is_queued_for_deletion(): continue
		if fx.get_script() == scripts.projectile: fx._physics_process(delta)
		elif fx.has_method("_process"): fx._process(delta)

func _case(kind: String) -> bool:

	var stage: Node2D = _stage()
	var target_stage: Node2D = _stage()

			source._spawn_hero_skill_fx("hua_shot", caster, foe.position, ad)
		"arrow_rain": source._spawn_hero_skill_fx("hua_rain", caster, foe.position, ad)

			_check("actual smite selects flameburst", source._spawn_smite_variant_fx(foe.position, 75.0, {"dot_total": 40.0}, ad))
		"ability": source._spawn_ability_fx(foe.position, 75.0, ad.color)

			# Actual death creator. Only unrelated empty-selection HUD fields are fixture nodes.
			source.phase = scripts.battle.Phase.DEPLOY
			source.hud = scripts.hud.new()
			source.hud.battle = source
			source.add_child(source.hud)
			for field: String in ["_info_name", "_info_hp", "_info_stats", "_port_char"]:
				var label := Label.new()
				source.hud.add_child(label)
				source.hud.set(field, label)
			source.hud._port_tex = TextureRect.new()
			source.hud.add_child(source.hud._port_tex)
			var building: Variant = _unit(source, 3, 0, Vector2(600, 300))
			building.key = "barracks"
			building.is_building = true
			building.hp = 0.0
			var texture: Texture2D = root.get_node("Art").building_texture("barracks")

			textures["building:barracks"] = texture
			source._on_unit_died(building)
			if not _check("actual building creator queues victim", building.is_queued_for_deletion()): return false

		var concurrent_arrow: Variant = scripts.projectile.new()
		concurrent_arrow.position = caster.position
		concurrent_arrow.setup(caster, foe, 11.0, false)
		source.fx_root.add_child(concurrent_arrow)

		concurrent_axes.caster = caster
		concurrent_axes.game = source
		concurrent_axes.dur = 0.9
		concurrent_axes.hits = [{"target": foe, "dmg": 17.0}]
		source.fx_root.add_child(concurrent_axes)
	# Warm actual visuals, then selected gameplay consumers. Attack remains before first pulse.
	var warm := 0.30 if kind == "arrow_shot_impact" else 0.08
	for fx: Variant in source.fx_root.get_children():
		if fx.has_method("_process"): fx._process(warm)

	contexts.append(new_visual)

	var expected_global_next: int = randi()
	seed(940133)
	var vp: Dictionary = new_visual.prepare(wire, CONTENT, up.id_to_unit, up.identity.expired_unit())
	if not _ok(kind + " real timed Fx JSON prepare", vp): return false

	_check(kind + " wrong owner count rejects before connection", not new_visual.bind_ground_fire_owner(target, target._ground_fire_visuals + 1).ok)

	_check(kind + " second bind refuses duplicate callback", not new_visual.bind_ground_fire_owner(target, target._ground_fire_visuals).ok)
	var new_meteor: Variant = _meteor_state(up.identity)
	var new_ground: Variant = scripts.run_continuous_effect_state.new(scripts.run_state_value_codec, scripts.battle, scripts.unit)
	if not _ok(kind + " meteor wards bind exact restored units", new_meteor.bind(target, JSON.parse_string(JSON.stringify(mc.record)), CONTENT, up.id_to_unit, up.identity.expired_unit())): return false
	if not _ok(kind + " ground bind exact restored units", new_ground.bind(target, JSON.parse_string(JSON.stringify(gc.record)), CONTENT, up.id_to_unit, up.identity.expired_unit())): return false
	up.identity.release_tombstones()
	new_visual.release_expired_fx()
	target_stage.add_child(target.fx_root)
	_check(kind + " distinct parents preserve exact FxRoot names", source.fx_root.get_parent() != target.fx_root.get_parent() and source.fx_root.name == &"FxRoot" and target.fx_root.name == &"FxRoot")
	if not _ok(kind + " paused visual activation", new_visual.activate()): return false
	_check(kind + " restore ready consumes no global visual RNG", randi() == expected_global_next)
	var target_ids: Dictionary = {}
	for id: String in up.id_to_unit: target_ids[up.id_to_unit[id]] = id
	var reread: Dictionary = new_visual.capture(target.fx_root, CONTENT, target_ids)
	_check(kind + " ready preserves all prior phase particles and texture", reread.ok and reread.value == vc.value)
	if kind == "building_collapse":
		_check("collapse restores same trusted Art AtlasTexture", is_same(target.fx_root.get_child(0).tex, textures["building:barracks"]))
	if kind == "ward_attack_boundary":
		var hp_before: float = up.id_to_unit["2"].hp
		seed(770000)
		source._ward_pass(0.55)
		seed(770000)
		target._ward_pass(0.55)
		var has_blink := false
		for fx: Variant in target.fx_root.get_children(): has_blink = has_blink or fx.get_script() == scripts.battle.BlinkShotFx
		_check("attack ward consumes damage once and creates real BlinkShotFx", up.id_to_unit["2"].hp == hp_before - 13.0 and has_blink)

		if not _ok("attack ward future mixed visual graph remains capturable", pulse_snapshot): return false
		var pulse_owner: Variant = _shell()
		var pulse_copy: Variant = _mixed_visual(pulse_owner)

		var pulse_prepared: Dictionary = pulse_copy.prepare(JSON.parse_string(JSON.stringify(pulse_snapshot.value)), CONTENT, up.id_to_unit)
		if not _ok("attack ward second generation JSON prepares every visual", pulse_prepared): return false
		pulse_owner.fx_root = pulse_prepared.root
		_stage().add_child(pulse_prepared.root)
		if not _ok("attack ward second generation activation", pulse_copy.activate()): return false
		var pulse_reread: Dictionary = pulse_copy.capture(pulse_prepared.root, CONTENT, target_ids)
		_check("attack ward second generation preserves complete captured visual records", pulse_reread.ok and pulse_reread.value == pulse_snapshot.value)
	var equal := true
	var disappearance_steps: Array = []
	var last_count: int = target.fx_root.get_child_count()
	for step: int in range(1, 66):
		# Future creation uses the same declared test-only global visual seed per pair.
		# This does not claim the engine's global visual RNG is a saved gameplay stream.
		seed(770000 + step)
		_step(source, 0.125)
		seed(770000 + step)
		_step(target, 0.125)
		await process_frame
		var ov: Dictionary = old_visual.capture(source.fx_root, CONTENT, ids)
		var nv: Dictionary = new_visual.capture(target.fx_root, CONTENT, target_ids)
		var om: Dictionary = old_meteor.capture(source, CONTENT, ids)
		var nm: Dictionary = new_meteor.capture(target, CONTENT, target_ids)
		var og: Dictionary = old_ground.capture(source, CONTENT, ids)
		var ng: Dictionary = new_ground.capture(target, CONTENT, target_ids)
		equal = equal and ov.ok and nv.ok and ov.value == nv.value and om.ok and nm.ok and om.record == nm.record and og.ok and ng.ok and og.record == ng.record
		equal = equal and caster.hp == up.id_to_unit["1"].hp and foe.hp == up.id_to_unit["2"].hp and source._ground_fire_visuals == target._ground_fire_visuals and source.capture_gameplay_rng().record == target.capture_gameplay_rng().record
		if target.fx_root.get_child_count() < last_count: disappearance_steps.append(step)
		last_count = target.fx_root.get_child_count()
	_check(kind + " 65 paired actual consumer and visual steps remain equal", equal)
	_check(kind + " all timed nodes authority and ground count expire", target.fx_root.get_child_count() == 0 and target._ground_fire_visuals == 0 and target._meteor_zones.is_empty() and target._wards.is_empty() and target._ground_dots.is_empty())
	_check(kind + " real disappearance observed", not disappearance_steps.is_empty())
	if kind == "ward_attack_boundary": _check("concurrent ward arrow and axes exact cumulative damage", foe.hp == 920.0 and up.id_to_unit["2"].hp == 920.0)
	_check(kind + " no gameplay RNG fault", source.gameplay_rng_fault().is_empty() and target.gameplay_rng_fault().is_empty())
	if kind == "meteor" or kind.begins_with("ground_fire"):
		_check(kind + " actual future damage consumed", foe.hp < 1000.0)
	if kind in ["ward_heal", "ward_loyalty"]: _check(kind + " actual future healing consumed", caster.hp > 500.0)
	observations[kind] = {"snapshot_nodes": vc.count, "disappearance_steps": disappearance_steps, "caster_hp": caster.hp, "foe_hp": foe.hp,
		"final_ground_count": target._ground_fire_visuals, "future_visual_randomness": "paired explicit test seeds; not full global RNG save"}
	return true

func _timed_negatives(kind: String, wire: Dictionary, textures: Dictionary, old_visual: Variant, ids: Dictionary) -> void:
	var codec: Variant = scripts.run_state_value_codec.new()
	var raw: Array = codec.decode(wire.records).value
	var known: Dictionary = {}
	for id: String in ids.values(): known[id] = true
	for mutation: String in ["unknown", "extra", "expired", "nonfinite", "particle", "texture", "texture_geometry", "ward_texture"]:
		var rows: Array = raw.duplicate(true)
		var row: Dictionary = rows[1]
		match mutation:
			"unknown": row.kind = "res://arbitrary.gd"
			"extra": row.values["surprise"] = 1
			"expired": row.values.t = 0.0
			"nonfinite": row.values.t = INF
			"particle":
				if row.values.has("_embers"): row.values._embers.pop_back()
				elif row.values.has("_arrows"): row.values._arrows[0]["missing_shape"] = 0.0
				else: continue
			"texture":
				if kind != "building_collapse": continue
				row.values.tex = {"state": "trusted", "key": "user://escape.png", "signature": {}}
			"texture_geometry":
				if kind != "building_collapse": continue
				row.values.tex.signature.width += 1
			"ward_texture":
				if row.kind != "ward": continue
				row.values.ward_visual = {"state": "missing_or_changed"}
		var enc: Dictionary = codec.encode(rows)
		if not enc.ok:
			_check(kind + " nonfinite rejected by unchanged codec", mutation == "nonfinite")
			continue
		var bad: Dictionary = wire.duplicate(true)
		bad.records = enc.value
		var validator: Variant = scripts.run_visual_graph.new(scripts.run_state_value_codec, scripts.battle, scripts.unit, null, textures)
		contexts.append(validator)
		_check(kind + " malformed timed graph rejects before allocation " + mutation, not validator.prepare(bad, CONTENT, {}).ok and validator._nodes.is_empty())
	_check(kind + " StringName still rejected by value codec", not codec.encode({&"chain_from": {"state": "none"}}).ok)
	if kind == "building_collapse":
		var missing: Variant = scripts.run_visual_graph.new(scripts.run_state_value_codec, scripts.battle, scripts.unit)
		contexts.append(missing)
		_check("missing trusted collapse texture rejects allocation", not missing.prepare(wire, CONTENT, {}).ok and missing._nodes.is_empty())
		var fake_root := Node2D.new()

	blink._S = Vector2(INF, 0)

	blink._S = original_s

	blink.material = null

	_check("blink adjacent unknown Fx rejects whole graph", old.capture(original, CONTENT, {}).get("code") == "UNSUPPORTED_VISUAL_SCRIPT")
	unknown.free()

	contexts.append(restore)
	seed(194021)
	var expected_next := randi()
	seed(194021)

	copy_stage.add_child(prepared.root)

	_check("blink restore ready consumes no global RNG", randi() == expected_next)
	for i: int in range(4): await process_frame

	if not _ok("blink one paused activation", restore.activate()): return false
	_check("blink duplicate activation refuses", restore.activate().get("code") == "VISUAL_ALREADY_ACTIVATED")
	var reread: Dictionary = restore.capture(prepared.root, CONTENT, {})
	_check("blink post-ready graph equals JSON source", reread.ok and reread.value == captured.value)
	var exits := {"original": 0, "restored": 0}
	blink.tree_exited.connect(func(): exits["original"] += 1)
	restored.tree_exited.connect(func(): exits["restored"] += 1)
	paused = false
	var equal := true
	var paired_live_frames := 0
	for i: int in range(100):
		await process_frame
		if is_instance_valid(blink) != is_instance_valid(restored): equal = false
		if not is_instance_valid(blink) and not is_instance_valid(restored): break
		if is_instance_valid(blink) and is_instance_valid(restored):
			paired_live_frames += 1
			equal = equal and blink.t == restored.t and blink._S == restored._S and blink._E == restored._E and blink.transform == restored.transform and blink.is_queued_for_deletion() == restored.is_queued_for_deletion()
	paused = true
	_check("blink real idle lifecycle stays equal through natural deletion", equal and paired_live_frames > 0 and not is_instance_valid(blink) and not is_instance_valid(restored))
	_check("blink both real exits fire exactly once", exits.original == 1 and exits.restored == 1)
	observations["blink_engine"] = {"paired_live_frames": paired_live_frames, "saved_remaining": saved_t, "exits": exits, "manual_process_calls": false, "cross_process": false, "rendered_pixels_verified": false}
	return failures.is_empty()
