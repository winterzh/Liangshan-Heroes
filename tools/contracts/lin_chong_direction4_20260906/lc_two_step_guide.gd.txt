## Two physically opposite strides, rendered directly from 3D primitives.
func rig(parent, alternate: bool) -> void:
	var gray=Color("454c54");var skin=Color("aa927a");var left=Color("d93830");var right=Color("245cd6")
	rod(parent,Vector3(0,0.85,0),Vector3(0,1.45,0),0.28,gray)
	sphere(parent,Vector3(0,1.70,-0.015),0.17,skin)
	sphere(parent,Vector3(0,1.98,0.01),0.07,Color("262626"))
	sphere(parent,Vector3(0,1.70,-0.18),0.055,skin)
	var rh=Vector3(0.42,1.10,-0.32)
	for arm in [[Vector3(-0.3,1.43,0),Vector3(-0.36,1.11,0.16),Vector3(-0.38,0.89,0.22),left],[Vector3(0.3,1.43,0),Vector3(0.47,1.17,-0.04),rh,right]]:
		rod(parent,arm[0],arm[1],0.085,arm[3]);rod(parent,arm[1],arm[2],0.07,arm[3]);sphere(parent,arm[2],0.085,arm[3])
	for side in [-1,1]:
		var front=side==1 if alternate else side==-1
		var x=side*0.16;var color=left if side==-1 else right
		var hip=Vector3(x,0.86,0)
		var knee=Vector3(x,0.65,-0.33) if front else Vector3(x,0.42,0.17)
		var foot=Vector3(x,0.20,-0.50) if front else Vector3(x,0.08,0.35)
		rod(parent,hip,knee,0.105,color);sphere(parent,knee,0.105,color);rod(parent,knee,foot,0.085,color);rod(parent,foot+Vector3(0,0,0.06),foot+Vector3(0,0,-0.16),0.08,color)
	var direction=Vector3(0,0.1,-1).normalized();var tip=rh+direction*1.1;var tail=rh-direction*0.7
	rod(parent,tail,tip,0.023,Color("b57625"));rod(parent,tip,tip+direction*0.23,0.044,Color("e0e4e7"))
