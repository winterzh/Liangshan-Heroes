func _initialize() -> void: _run.call_deferred()

func check(ok: bool, label: String) -> void:
	checks.append({"label": label, "passed": ok})
	if not ok: failures.append(label)

func _normalize(path: String) -> String:
	return path.replace("\\", "/").simplify_path().trim_suffix("/")

func _source_pins() -> Dictionary:
	var result: Dictionary = {}
	for path in SOURCE_PATHS:
		result[path] = FileAccess.get_sha256("res://" + path)
		check(result[path].length() == 64, "source raw SHA exists: " + path)
	return result

func _hop(record: Dictionary) -> Dictionary:
	var parser := JSON.new()
	var error: int = parser.parse(JSON.stringify(record, "", false))
	check(error == OK and typeof(parser.data) == TYPE_DICTIONARY, "record crosses real JSON syntax")
	return parser.data if error == OK and typeof(parser.data) == TYPE_DICTIONARY else {}

func _repack(record: Dictionary, mutate: Callable) -> Dictionary:
	var decoded: Dictionary = codec.decode(record.values)
	check(decoded.ok, "negative fixture starts from actual decoded values")
	if not decoded.ok: return {}
	mutate.call(decoded.value)
	var encoded: Dictionary = codec.encode(decoded.value)
	check(encoded.ok, "negative fixture encoded with real value codec")
	if not encoded.ok: return {}
	var changed: Dictionary = record.duplicate(true)
	changed.values = encoded.value
	return _hop(changed)

func _reject(record: Variant, code: String, label: String) -> void:
	var result: Dictionary = adapter.validate_record(record, CONTENT_FIXTURE)
	check(not result.ok and result.code == code and not result.has("values") and not result.has("record") \
		and result.restore_ready == false, label)

func _run() -> void:
	var expected_user: String = OS.get_environment("INVENTORY_QA_EXPECTED_USER_DIR")
	if expected_user.is_empty() or _normalize(expected_user) != _normalize(OS.get_user_data_dir()):
		push_error("Inventory QA requires the root-owned exact private user directory")
		quit(2)
		return
	source_before = _source_pins()
	inventory_script = load("res://scripts/hero_inventory.gd") as GDScript
	var codec_script: GDScript = load("res://scripts/run_state_value_codec.gd") as GDScript
	var adapter_script: GDScript = load("res://scratchpad/run_inventory_state/inventory_values.gd") as GDScript
	check(inventory_script != null and codec_script != null and adapter_script != null, "real inventory codec and draft scripts loaded")
	if inventory_script == null or codec_script == null or adapter_script == null:
		quit(2)
		return
	codec = codec_script.new()
	adapter = adapter_script.new(inventory_script, codec_script)
	_cases()
	var source_after: Dictionary = _source_pins()
	check(source_before == source_after, "five runtime source files unchanged")
	var report: Dictionary = {"schema": 1, "passed": not checks.is_empty() and failures.is_empty(),
		"checks": checks, "check_count": checks.size(), "failures": failures,
		"process_id": OS.get_process_id(), "actual_user_dir": OS.get_user_data_dir(),
		"engine": Engine.get_version_info(), "source_raw_sha256": source_before,
		"real_inventory_object": true, "capture_validate_only": true, "owner_is_test_probe": true,
		"assignment_restore_implemented": false, "business_definitions_validated": false,
		"global_item_uid_allocator_restored": false, "battle_resume_tested": false}
	var output: String = OS.get_environment("INVENTORY_QA_OUT")
	if not output.is_empty():
		var resolved: String = _normalize(ProjectSettings.globalize_path(output))
		if not resolved.begins_with(_normalize(OS.get_user_data_dir()) + "/") or FileAccess.file_exists(resolved):
			push_error("Inventory QA output must be a new file within the supplied private user directory")
			quit(2)
			return
		var file := FileAccess.open(resolved, FileAccess.WRITE)
		if file == null:
			push_error("Cannot create inventory QA report")
			quit(2)
			return
		var stored: bool = file.store_string(JSON.stringify(report, "\t", false) + "\n")
		file.flush()
		var error: int = file.get_error()
		file.close()
		if not stored or error != OK:
			push_error("Incomplete inventory QA report write")
			quit(2)
			return
	print("[inventory-state QA] " + JSON.stringify(report, "", false))
	quit(0 if report.passed else 1)
