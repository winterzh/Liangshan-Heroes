## Deterministic 3D anatomy reference only, never a production sprite.
func mat(color: Color) -> StandardMaterial3D:
	var m=StandardMaterial3D.new();m.albedo_color=color;m.roughness=0.8;return m
func sphere(parent,point:Vector3,radius:float,color:Color) -> void:
	var n=MeshInstance3D.new();var mesh=SphereMesh.new();mesh.radius=radius;mesh.height=radius*2
	n.mesh=mesh;n.material_override=mat(color);n.position=point;parent.add_child(n)
func rod(parent,a:Vector3,b:Vector3,radius:float,color:Color) -> void:
	var n=MeshInstance3D.new();var mesh=CylinderMesh.new();mesh.top_radius=radius;mesh.bottom_radius=radius;mesh.height=a.distance_to(b)
	n.mesh=mesh;n.material_override=mat(color);n.position=(a+b)*0.5;n.quaternion=Quaternion(Vector3.UP,(b-a).normalized());parent.add_child(n)
func _initialize() -> void: _run.call_deferred()
