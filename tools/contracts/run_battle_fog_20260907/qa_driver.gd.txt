	"scratchpad/run_battle_fog_state/fog_values.gd", "scratchpad/run_battle_fog_state/qa_driver.gd"]
class WrongObject extends RefCounted:
	var calls := 0
	func _to_string() -> String:
		calls += 1
		return "not a Battle"
	func _get_property_list() -> Array[Dictionary]:
		calls += 1
		return []

func _initialize() -> void: _run.call_deferred()

func check(ok: bool, label: String, group := "behavior") -> void:
	checks.append({"label": label, "passed": ok, "group": group})
	if not ok: failures.append(label)

func _normalize(path: String) -> String:
	return path.replace("\\", "/").simplify_path().trim_suffix("/")

func _source_pins(expected: Dictionary, prefix: String) -> Dictionary:
	var result: Dictionary = {}
	for path in SOURCE_PATHS:
		result[path] = FileAccess.get_sha256("res://" + path)
		check(expected.has(path) and result[path].length() == 64 and result[path] == expected.get(path, ""), prefix + path, "source")
	return result

func _hop(record: Dictionary) -> Dictionary:
	var parser := JSON.new()
	var error: int = parser.parse(JSON.stringify(record, "", false))
	check(error == OK and typeof(parser.data) == TYPE_DICTIONARY, "record crosses real JSON syntax", "json")
	return parser.data if error == OK and typeof(parser.data) == TYPE_DICTIONARY else {}

func _f64(value: float) -> String:
	var raw := PackedByteArray()
	raw.resize(8)
	raw.encode_double(0, value)
	return raw.hex_encode()

func _f32(values: PackedFloat32Array) -> String:
	var raw := PackedByteArray()
	raw.resize(values.size() * 4)
	for index in range(values.size()): raw.encode_float(index * 4, values[index])
	return raw.hex_encode()

func _populate(battle, actual_map, width: int, height: int) -> void:
	# These assignments only construct QA input. No loaded record is applied.
	actual_map.w = width
	actual_map.h = height
	battle.fog = true
	battle._vision = PackedByteArray()
	battle._sight_now = PackedByteArray()
	battle._reveal_t = PackedFloat32Array()
	battle._vision.resize(width * height)
	battle._sight_now.resize(width * height)
	battle._reveal_t.resize(width * height)
	for index in range(width * height):
		battle._vision[index] = index % 3
		battle._sight_now[index] = 1 if index % 3 == 2 else 0
		battle._reveal_t[index] = 0.1 if index % 11 == 0 else 0.0
		if battle._reveal_t[index] > 0.0: battle._vision[index] = 2
	battle._fog_t = 0.123456789012345

func _repack(record: Dictionary, mutate: Callable) -> Dictionary:
	var decoded: Dictionary = codec.decode(record.values)
	check(decoded.ok, "negative fixture decodes valid original value tree", "fixture")
	if not decoded.ok: return {}
	mutate.call(decoded.value)
	var encoded: Dictionary = codec.encode(decoded.value)
	check(encoded.ok, "negative fixture encoded with the real value codec", "fixture")
	if not encoded.ok: return {}
	var changed: Dictionary = record.duplicate(true)
	changed.values = encoded.value
	return _hop(changed)

func _reject(record: Variant, code: String, label: String, dimensions := Vector2i(7, 3)) -> void:
	var result: Dictionary = adapter.validate_record(record, CONTENT_FIXTURE, dimensions)
	check(not result.ok and result.code == code and not result.has("values") and not result.has("record") \
		and result.restore_ready == false, label, "rejection")

func _capture_reject(subject: Variant, code: String, label: String) -> void:
	var result: Dictionary = adapter.capture_values(subject, CONTENT_FIXTURE)
	check(not result.ok and result.code == code and not result.has("values") and not result.has("record") \
		and result.restore_ready == false, label, "rejection")

func _replace_tag(record: Dictionary, field: String, value: Dictionary) -> Dictionary:
	var changed: Dictionary = record.duplicate(true)
	var found := false
	for entry in changed.values.entries:
		if entry[0].t == "string" and entry[0].v == field:
			entry[1] = value
			found = true
	check(found, "encoded field replacement identifies exact key", "fixture")
	return changed

func _run() -> void:
	var manifest_path: String = OS.get_environment("BATTLE_FOG_QA_MANIFEST")
	if manifest_path.is_empty() or not FileAccess.file_exists(manifest_path):
		push_error("Battle fog QA requires a root-owned private manifest")
		quit(2)
		return
	var input := FileAccess.open(manifest_path, FileAccess.READ)
	if input == null or input.get_length() < 2 or input.get_length() > 65536:
		push_error("Battle fog QA manifest size or open rejected")
		quit(2)
		return
	var text: String = input.get_as_text()
	input.close()
	var parser := JSON.new()
	if parser.parse(text) != OK or typeof(parser.data) != TYPE_DICTIONARY:
		push_error("Battle fog QA manifest must be a JSON Dictionary")
		quit(2)
		return
	var manifest: Dictionary = parser.data
	for field in ["run_id", "run_dir", "private_user", "report"]:
		if not manifest.has(field) or typeof(manifest[field]) != TYPE_STRING or manifest[field].is_empty():
			push_error("Battle fog QA manifest missing required String")
			quit(2)
			return
	if not manifest.has("source_sha256") or typeof(manifest.source_sha256) != TYPE_DICTIONARY:
		push_error("Battle fog QA manifest missing source hashes")
		quit(2)
		return
	var run_dir: String = _normalize(manifest.run_dir)
	var report_path: String = _normalize(manifest.report)
	var private_user: String = _normalize(manifest.private_user)
	var runs_dir: String = _normalize(ProjectSettings.globalize_path("res://scratchpad/run_battle_fog_state/runs"))
	if run_dir.get_base_dir() != runs_dir or run_dir.get_file() != manifest.run_id \
			or _normalize(manifest_path) != run_dir + "/manifest.json" or report_path != run_dir + "/report.json" \
			or not private_user.begins_with(run_dir + "/private_profile/") or FileAccess.file_exists(report_path):
		push_error("Battle fog QA paths must remain in the new root-owned run")
		quit(2)
		return
	var manifest_sha: String = FileAccess.get_sha256(manifest_path)
	check(_normalize(OS.get_user_data_dir()) == private_user, "parent-isolated actual profile", "environment")
	check(manifest.source_sha256.size() == SOURCE_PATHS.size(), "exact five runtime source entries", "source")
	var before: Dictionary = _source_pins(manifest.source_sha256, "runtime source ")
	if not failures.is_empty():
		push_error("Battle fog QA source or actual private environment mismatch")
		quit(2)
		return
	battle_script = load("res://scripts/battle.gd") as GDScript
	map_script = load("res://scripts/game_map.gd") as GDScript
	codec_script = load("res://scripts/run_state_value_codec.gd") as GDScript
	adapter_script = load("res://scratchpad/run_battle_fog_state/fog_values.gd") as GDScript
	check(battle_script != null and map_script != null and codec_script != null and adapter_script != null,
		"real Battle map codec and adapter scripts loaded", "environment")
	if battle_script == null or map_script == null or codec_script == null or adapter_script == null:
		quit(2)
		return
	codec = codec_script.new()
	adapter = adapter_script.new(battle_script, map_script, codec_script)
	_cases()
	var after: Dictionary = _source_pins(manifest.source_sha256, "source unchanged ")
	check(before == after, "five runtime source files unchanged", "source")
	check(FileAccess.get_sha256(manifest_path) == manifest_sha, "root manifest raw bytes unchanged", "source")
	var groups: Dictionary = {}
	for row in checks: groups[row.group] = int(groups.get(row.group, 0)) + 1
	var report: Dictionary = {"schema": 1, "run_id": manifest.run_id, "manifest_sha256": manifest_sha,
		"passed": not checks.is_empty() and failures.is_empty(),
		"checks": checks, "check_count": checks.size(), "check_groups": groups, "failures": failures,
		"process_id": OS.get_process_id(), "actual_user_dir": OS.get_user_data_dir(), "engine": Engine.get_version_info(),
		"source_sha256": before, "source_after_sha256": after, "real_battle_and_map_objects": true, "objects_outside_simulation_tree": true,
		"capture_validate_only": true, "assignment_restore_implemented": false, "render_resources_rebuilt": false,
		"snapshot_barrier_implemented": false, "unit_visibility_reconciled": false, "battle_resume_tested": false,
		"restore_ready": false, "coverage": adapter.coverage()}
	if FileAccess.file_exists(report_path):
		push_error("Battle fog QA refuses an existing report")
		quit(2)
		return
	var file := FileAccess.open(report_path, FileAccess.WRITE)
	if file == null:
		push_error("Cannot create battle fog QA report")
		quit(2)
		return
	var stored: bool = file.store_string(JSON.stringify(report, "\t", false) + "\n")
	file.flush()
	var error: int = file.get_error()
	file.close()
	if not stored or error != OK:
